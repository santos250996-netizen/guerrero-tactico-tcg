   207→   207→function freshHandFromPile(pile: HandCard[], n: number): HandCard[] {
   208→   208→  const out: HandCard[] = [];
   209→   209→  for (let i = 0; i < n && pile.length > 0; i++) {
   210→   210→    const card = pile.pop()!;
   211→   211→    out.push({ ...card, uid: uuidv4(), handBonus: 0 });
   212→   212→  }
   213→   213→  // ensureCost1 guarantee: if the starting hand has no cost-1 card, swap in
   214→   214→  // one from the pile (so the player can always play something on turn 1).
   215→   215→  if (!out.some((c) => c.cost === 1) && pile.length > 0) {
   216→   216→    const idx = pile.findIndex((c) => c.cost === 1);
   217→   217→    if (idx >= 0) {
   218→   218→      const replacement = pile.splice(idx, 1)[0];
   219→   219→      // Swap out the highest-cost card to make room
   220→   220→      let swapIdx = 0;
   221→   221→      for (let i = 1; i < out.length; i++) {
   222→   222→        if (out[i].cost > out[swapIdx].cost) swapIdx = i;
   223→   223→      }
   224→   224→      pile.push(out[swapIdx]);
   225→   225→      out[swapIdx] = { ...replacement, uid: uuidv4(), handBonus: 0 };
   226→   226→    }
   227→   227→  }
   228→   228→  return out;
   229→   229→}

   268→   268→/**
   269→   269→ * Resolve the player's deck from the loaded card pool + deck card IDs.
   270→   270→ * Mago/Asesino siempre usan su mazo espejo; Guerrero permite custom de BD.
   271→   271→ */
   272→   272→function resolveDeck(state: GameStoreState): CardDef[] {
   273→   273→  if (state.gameClass !== "guerrero") {
   274→   274→    return CLASS_DECKS[state.gameClass];
   275→   275→  }
   276→   276→  const { cardPool, deckCardIds } = state;
   277→   277→  if (cardPool.length > 0 && deckCardIds.length > 0) {
   278→   278→    const resolved = deckCardIds
   279→   279→      .map((id) => cardPool.find((c) => c.id === id))
   280→   280→      .filter((c): c is CardDef => Boolean(c));
   281→   281→    if (resolved.length === deckCardIds.length) {
   282→   282→      return resolved;
   283→   283→    }
   284→   284→  }
   285→   285→  return WARRIOR_DECK;
   286→   286→}

   288→   288→/**

   293→   293→export function flashLaneBonuses(bonuses: LaneBonuses) {
   294→   294→  useGameStore.setState({ laneBonusFlashes: { ...bonuses } });

   297→   297→      laneBonusFlashes: nullBonuses(ALL_LANE_IDS),
   298→   298→    });
   299→   299→  }, 1500);
   300→   300→}

   410→   410→    if (getFinalCost(baseCost, laneBonus) > energy) return false;

   417→   417→    if (state.phase !== "playing") return;
   418→   418→    if (!state.selectedCardUid) return;
   419→   419→
   420→   420→    const card = state.hand.find((c) => c.uid === state.selectedCardUid);
   421→   421→    if (!card) return;
   422→   422→
   423→   423→    const laneBonus = state.laneBonuses[lane];
   424→   424→    const activeTrap = state.laneTraps[lane];
   425→   425→
   426→   426→    // Pre-flight checks with friendly messages (before playCard throws)
   427→   427→    if (!isCardCostAllowed(card.cost, laneBonus)) {
   428→   428→      set({
   429→   429→        invalidPlayMsg: `"${card.name}" no se puede jugar en ${lane}: el bono exige coste exacto.`,
   430→   430→      });
   431→   431→      return;
   432→   432→    }
   433→   433→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   434→   434→    const effCost = getFinalCost(baseCost, laneBonus);
   435→   435→    if (effCost > state.energy) {
   436→   436→      set({
   437→   437→        invalidPlayMsg: `Sin energía suficiente para "${card.name}" (necesita ${effCost}${laneBonus?.kind === "cost_neg" || laneBonus?.kind === "cost_pos" ? " ajustado por bono" : ""}, tienes ${state.energy}).`,
   438→   438→      });
   439→   439→      return;
   440→   440→    }
   441→   441→    const maxCards = getLaneMaxCards(laneBonus);
   442→   442→    if (state.board[lane].player.length >= maxCards) {
   443→   443→      set({
   444→   444→        invalidPlayMsg: `El carril ${lane} está lleno (máx. ${maxCards} por bando).`,
   445→   445→      });
   446→   446→      return;
   447→   447→    }
   448→   448→
   449→   449→    try {
   450→   450→      // Capture undo snapshot BEFORE the play
   451→   451→      const undoSnapshot: UndoSnapshot = {
   452→   452→        board: state.board,
   453→   453→        hand: state.hand,
   454→   454→        energy: state.energy,
   455→   455→        laneTraps: state.laneTraps,
   456→   456→        nextTurnEnergyPenalty: state.nextTurnEnergyPenalty,
   457→   457→        gameStats: { ...state.gameStats },
   458→   458→        playsThisTurn: state.playsThisTurn,
   459→   459→      };
   460→   460→
   461→   461→      const res = playCard(
   462→   462→        state.board,
   463→   463→        state.hand,
   464→   464→        state.energy,
   465→   465→        "player",
   466→   466→        lane,
   467→   467→        card,
   468→   468→        laneBonus,
   469→   469→        activeTrap,
   470→   470→        state.playsThisTurn + 1
   471→   471→      );
   472→   472→
   473→   473→      // Sounds: card play, then buff/debuff/remove/trap as appropriate
   474→   474→      if (useSettings.getState().soundOn) {
   475→   475→        sfx.cardPlay();
   476→   476→        const hasBuff = res.mutations.some((m) => m.powerDelta > 0);
   477→   477→        const hasDebuff = res.mutations.some((m) => m.powerDelta < 0);
   478→   478→        const hasRemove = res.removedCards.length > 0;
   479→   479→        if (hasBuff) sfx.buff();
   480→   480→        if (hasDebuff) sfx.debuff();
   481→   481→        if (hasRemove) sfx.remove();
   482→   482→        if (res.trapTriggered) sfx.trapTrigger();
   483→   483→      }
   484→   484→
   485→   485→      // Floating numbers from mutations
   486→   486→      const floatingNumbers: FloatingNumber[] = res.mutations.map((m) => ({
   487→   487→        id: uuidv4(),
   488→   488→        uid: m.uid,
   489→   489→        value: m.powerDelta,
   490→   490→      }));
   491→   491→
   492→   492→      // Track game stats

  1015→  1015→  swapCardInDeck: (index, cardId) => {
  1016→  1016→    // Allow placing a card into ANY of the 12 deck slots, even if the slot is
  1017→  1017→    // beyond the current deckCardIds length (e.g. user emptied the deck then
  1018→  1018→    // clicks slot 5 — we pad with empty strings up to that index).
  1019→  1019→    if (index < 0 || index >= 12) return;
  1020→  1020→    const { deckCardIds } = get();
  1021→  1021→    if (deckCardIds[index] === cardId) return;
  1022→  1022→    // Prevent duplicates (deck is 12 unique cards) — if the card is already in
  1023→  1023→    // the deck at a DIFFERENT slot, move it from the old slot to the new one
  1024→  1024→    // (old slot becomes empty). This is the "drag-to-rearrange" UX.
  1025→  1025→    const next = [...deckCardIds];
  1026→  1026→    // Pad with empty strings up to the target index so we can place at any slot
  1027→  1027→    while (next.length <= index) next.push("");
  1028→  1028→    if (deckCardIds.includes(cardId)) {
  1029→  1029→      const existingIdx = deckCardIds.indexOf(cardId);
  1030→  1030→      next[existingIdx] = "";  // clear old slot
  1031→  1031→    }
  1032→  1032→    next[index] = cardId;
  1033→  1033→    // Always store exactly 12 entries (pad/truncate to 12)
  1034→  1034→    while (next.length < 12) next.push("");
  1035→  1035→    set({ deckCardIds: next.slice(0, 12), deckDirty: true, deckSaveStatus: "idle" });
  1036→  1036→  },
