   197→function freshHandFromPile(pile: HandCard[], n: number): HandCard[] {
   198→  const out: HandCard[] = [];
   199→  for (let i = 0; i < n && pile.length > 0; i++) {
   200→    const card = pile.pop()!;
   201→    out.push({ ...card, uid: uuidv4(), handBonus: 0 });
   202→  }
   203→  // ensureCost1 guarantee: if the starting hand has no cost-1 card, swap in
   204→  // one from the pile (so the player can always play something on turn 1).
   205→  if (!out.some((c) => c.cost === 1) && pile.length > 0) {
   206→    const idx = pile.findIndex((c) => c.cost === 1);
   207→    if (idx >= 0) {
   208→      const replacement = pile.splice(idx, 1)[0];
   209→      // Swap out the highest-cost card to make room
   210→      let swapIdx = 0;
   211→      for (let i = 1; i < out.length; i++) {
   212→        if (out[i].cost > out[swapIdx].cost) swapIdx = i;
   213→      }
   214→      pile.push(out[swapIdx]);
   215→      out[swapIdx] = { ...replacement, uid: uuidv4(), handBonus: 0 };
   216→    }
   217→  }
   218→  return out;
   219→}

   221→function pushLog(
   222→  list: LogEntry[],
   223→  turn: number,
   224→  side: LogEntry["side"],
   225→  text: string,
   226→  kind: LogEntry["kind"] = "info"
   227→): LogEntry[] {
   228→  const entry: LogEntry = {
   229→    id: uuidv4(),
   230→    turn,
   231→    side,
   232→    text,
   233→    kind,
   234→  };
   235→  return [entry, ...list].slice(0, MAX_LOG_ENTRIES);
   236→}

   238→/**
   239→ * Push all log entries from a TriggerResult onto the game log.
   240→ * Used by the turn-lifecycle hooks (start_turn, end_turn, on_death, umbral).
   241→ */
   242→function pushTriggerLogs(
   243→  list: LogEntry[],
   244→  turn: number,
   245→  result: { logs: TriggerResult["logs"] }
   246→): LogEntry[] {
   247→  let next = list;
   248→  for (const l of result.logs) {
   249→    next = pushLog(next, turn, l.side, l.text, "info");
   250→  }
   251→  return next;
   252→}

   258→/**
   259→ * Resolve the player's deck from the loaded card pool + deck card IDs.
   260→ * Falls back to WARRIOR_DECK if the pool/deck isn't loaded or has missing cards.
   261→ */
   262→function resolveDeck(state: GameStoreState): CardDef[] {
   263→  const { cardPool, deckCardIds } = state;
   264→  if (cardPool.length > 0 && deckCardIds.length > 0) {
   265→    const resolved = deckCardIds
   266→      .map((id) => cardPool.find((c) => c.id === id))
   267→      .filter((c): c is CardDef => Boolean(c));
   268→    if (resolved.length === deckCardIds.length) {
   269→      return resolved;
   270→    }
   271→  }
   272→  return WARRIOR_DECK;
   273→}

   289→export const useGameStore = create<GameStoreState>((set, get) => ({
   290→  turn: 1,
   291→  energy: 1,
   292→  maxEnergy: 1,
   293→  playsThisTurn: 0,
   294→  phase: "playing",
   295→  result: null,
   296→  finalResult: null,
   297→  drawPile: [],
   298→  hand: [],
   299→  board: emptyBoard(),
   300→  laneBonuses: { armas: null, proteccion: null, furia: null },
   301→  laneBonusFlashes: { armas: null, proteccion: null, furia: null },
   302→  laneTraps: emptyLaneTraps(),
   303→  nextTurnEnergyPenalty: 0,
   304→  floatingNumbers: [],
   305→  aiTurnPhase: "idle",
   306→  previewCard: null,
   307→  gameStats: emptyGameStats(),
   308→  undoSnapshot: null,
   309→  legendaryBurst: false,
   310→  view: "main-menu",
   311→  cardPool: [],
   312→  deckCardIds: [],
   313→  deckDirty: false,
   314→  deckSaveStatus: "idle",
   315→  deckLoading: false,
   316→  deckEditorReturn: "main-menu",
   317→  selectedCardUid: null,
   318→  draggingCardUid: null,
   319→  invalidPlayMsg: null,
   320→  lastEnemyPlays: [],
   321→  log: [],
   322→
   323→  initGame: () => {

   325→    const pile = buildDrawPileFrom(resolvedDeck).map((c) => ({
   326→      ...c,
   327→      uid: "",
   328→      handBonus: 0,
   329→    }));
   330→    const drawPile = pile.map((c) => ({ ...c, uid: uuidv4() }));
   331→    const hand = freshHandFromPile(drawPile, STARTING_HAND_SIZE);
   332→    const laneBonuses = rollLaneBonuses();
   333→    const bonusSummary = LANE_IDS.map((ln) => {
   334→      const b = laneBonuses[ln];
   335→      return b ? `${ln}: ${b.label}` : `${ln}: sin bono`;
   336→    }).join(" · ");
   337→    set({
   338→      turn: 1,
   339→      energy: 1,
   340→      maxEnergy: 1,
   341→      playsThisTurn: 0,
   342→      phase: "playing",
   343→      result: null,
   344→      finalResult: null,
   345→      drawPile,
   346→      hand,
   347→      board: emptyBoard(),
   348→      laneBonuses,
   349→      laneTraps: emptyLaneTraps(),
   350→      nextTurnEnergyPenalty: 0,
   351→      floatingNumbers: [],
   352→      aiTurnPhase: "idle",
   353→      previewCard: null,
   354→      gameStats: emptyGameStats(),
   355→      undoSnapshot: null,
   356→      legendaryBurst: false,
   357→      selectedCardUid: null,
   358→      invalidPlayMsg: null,
   359→      lastEnemyPlays: [],
   360→      log: pushLog(
   361→        pushLog([], 1, "system", "¡Comienza la batalla! Tienes 6 turnos.", "info"),
   362→        1,
   363→        "system",
   364→        `Bonos de carril: ${bonusSummary}.`,
   365→        "info"
   366→      ),
   367→    });
   368→    // Trigger visual flash on each lane showing its bonus
   369→    flashLaneBonuses(laneBonuses);
   370→  },
   371→
   372→  selectCard: (uid) => {
   373→    const cur = get().selectedCardUid;
   374→    set({ selectedCardUid: cur === uid ? null : uid, invalidPlayMsg: null });
   375→    if (useSettings.getState().soundOn) sfx.select();
   376→  },
   377→
   378→  setDraggingCard: (uid) => {
   379→    set({ draggingCardUid: uid });
   380→  },
   381→
   382→  canPlaySelected: (lane) => {
   383→    const { selectedCardUid, hand, board, energy, laneBonuses } = get();
   384→    if (!selectedCardUid) return false;
   385→    const card = hand.find((c) => c.uid === selectedCardUid);
   386→    if (!card) return false;
   387→    const laneBonus = laneBonuses[lane];
   388→    if (!isCardCostAllowed(card.cost, laneBonus)) return false;
   389→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   390→    if (getFinalCost(baseCost, laneBonus) > energy) return false;
   391→    if (board[lane].player.length >= getLaneMaxCards(laneBonus)) return false;
   392→    return true;
   393→  },
   394→
   395→  playSelectedCardIntoLane: (lane) => {
   396→    const state = get();
   397→    if (state.phase !== "playing") return;
   398→    if (!state.selectedCardUid) return;
   399→
   400→    const card = state.hand.find((c) => c.uid === state.selectedCardUid);
   401→    if (!card) return;
   402→
   403→    const laneBonus = state.laneBonuses[lane];
   404→    const activeTrap = state.laneTraps[lane];
   405→
   406→    // Pre-flight checks with friendly messages (before playCard throws)
   407→    if (!isCardCostAllowed(card.cost, laneBonus)) {
   408→      set({
   409→        invalidPlayMsg: `"${card.name}" no se puede jugar en ${lane}: el bono exige coste exacto.`,
   410→      });
   411→      return;
   412→    }
   413→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   414→    const effCost = getFinalCost(baseCost, laneBonus);
   415→    if (effCost > state.energy) {
   416→      set({
   417→        invalidPlayMsg: `Sin energía suficiente para "${card.name}" (necesita ${effCost}${laneBonus?.kind === "cost_neg" || laneBonus?.kind === "cost_pos" ? " ajustado por bono" : ""}, tienes ${state.energy}).`,
   418→      });
   419→      return;
   420→    }
   421→    const maxCards = getLaneMaxCards(laneBonus);
   422→    if (state.board[lane].player.length >= maxCards) {
   423→      set({
   424→        invalidPlayMsg: `El carril ${lane} está lleno (máx. ${maxCards} por bando).`,
   425→      });
   426→      return;
   427→    }
   428→
   429→    try {
   430→      // Capture undo snapshot BEFORE the play
   431→      const undoSnapshot: UndoSnapshot = {
   432→        board: state.board,
   433→        hand: state.hand,
   434→        energy: state.energy,
   435→        laneTraps: state.laneTraps,
   436→        nextTurnEnergyPenalty: state.nextTurnEnergyPenalty,
   437→        gameStats: { ...state.gameStats },
   438→        playsThisTurn: state.playsThisTurn,
   439→      };
   440→
   441→      const res = playCard(
   442→        state.board,
   443→        state.hand,
   444→        state.energy,
   445→        "player",
   446→        lane,
   447→        card,
   448→        laneBonus,
   449→        activeTrap,
   450→        state.playsThisTurn + 1
   451→      );
   452→
   453→      // Sounds: card play, then buff/debuff/remove/trap as appropriate
   454→      if (useSettings.getState().soundOn) {
   455→        sfx.cardPlay();
   456→        const hasBuff = res.mutations.some((m) => m.powerDelta > 0);
   457→        const hasDebuff = res.mutations.some((m) => m.powerDelta < 0);
   458→        const hasRemove = res.removedCards.length > 0;
   459→        if (hasBuff) sfx.buff();
   460→        if (hasDebuff) sfx.debuff();
   461→        if (hasRemove) sfx.remove();
   462→        if (res.trapTriggered) sfx.trapTrigger();
   463→      }
   464→
   465→      // Floating numbers from mutations
   466→      const floatingNumbers: FloatingNumber[] = res.mutations.map((m) => ({
   467→        id: uuidv4(),
   468→        uid: m.uid,
   469→        value: m.powerDelta,
   470→      }));
   471→
   472→      // Track game stats
   473→      const stats: GameStats = { ...state.gameStats };
   474→      stats.cardsPlayedByPlayer++;
   475→      stats.buffsApplied += res.mutations.filter((m) => m.powerDelta > 0).length;
   476→      stats.debuffsToEnemy += res.mutations.filter(
   477→        (m) => m.powerDelta < 0 && m.side === "enemy"
   478→      ).length;
   479→      stats.enemyCardsDestroyed += res.removedCards.filter(
   480→        (rc) => rc.side === "enemy"
   481→      ).length;
   482→      stats.totalEnergyUsed += effCost;
   483→      // Lane turnaround: was losing or tied, now winning
   484→      const oldPlayerPower = state.board[lane].player.reduce(
   485→        (a, c) => a + c.currentPower,
   486→        0
   487→      );
   488→      const oldEnemyPower = state.board[lane].enemy.reduce(
   489→        (a, c) => a + c.currentPower,
   490→        0
   491→      );
   492→      const newPlayerPower = res.board[lane].player.reduce(
   493→        (a, c) => a + c.currentPower,
   494→        0
   495→      );
   496→      const newEnemyPower = res.board[lane].enemy.reduce(
   497→        (a, c) => a + c.currentPower,
   498→        0
   499→      );
   500→      if (oldPlayerPower <= oldEnemyPower && newPlayerPower > newEnemyPower) {
   501→        stats.laneTurnarounds++;
   502→      }
   503→
   504→      // Update lane traps: clear if triggered (consumed), set if newTrap
   505→      let newLaneTraps: LaneTraps = state.laneTraps;
   506→      if (res.trapTriggered) {
   507→        newLaneTraps = { ...newLaneTraps, [lane]: null };
   508→      }
   509→      if (res.newTrap) {
   510→        newLaneTraps = { ...newLaneTraps, [lane]: res.newTrap };
   511→      }
   512→
   513→      // Energy penalty for next turn (Sobrecarga)
   514→      let nextTurnEnergyPenalty = state.nextTurnEnergyPenalty;
   515→      if (res.energyNextTurnDelta) {
   516→        nextTurnEnergyPenalty += res.energyNextTurnDelta;
   517→      }
   518→
   519→      let log = pushLog(
   520→        state.log,
   521→        state.turn,
   522→        "player",
   523→        `Juega "${card.name}" en ${lane} (Poder ${res.playedCard.currentPower}).`,
   524→        "play"
   525→      );
   526→      for (const m of res.mutations) {
   527→        // Shield-only mutations (Amparo) log as "Escudo +X" instead of "Debuff 0"
   528→        if (m.powerDelta === 0 && m.shieldDelta !== undefined && m.shieldDelta !== 0) {
   529→          log = pushLog(
   530→            log,
   531→            state.turn,
   532→            m.side === "player" ? "player" : "enemy",
   533→            `Escudo +${m.shieldDelta} en ${m.lane}.`,
   534→            "buff"
   535→          );
   536→          continue;
   537→        }
   538→        const verb = m.powerDelta > 0 ? "buff" : "debuff";
   539→        const sign = m.powerDelta > 0 ? "+" : "";
   540→        log = pushLog(
   541→          log,
   542→          state.turn,
   543→          m.side === "player" ? "player" : "enemy",
   544→          `${verb === "buff" ? "Buff" : "Debuff"} ${sign}${m.powerDelta} en ${m.lane}.`,
   545→          verb as "buff" | "debuff"
   546→        );
   547→      }
   548→      for (const rc of res.removedCards) {
   549→        log = pushLog(
   550→          log,
   551→          state.turn,
   552→          rc.side === "player" ? "player" : "enemy",
   553→          `Carta destruida en ${rc.lane} (${rc.reason}).`,
   554→          "debuff"
   555→        );
   556→      }
   557→      if (res.trapTriggered) {
   558→        log = pushLog(
   559→          log,
   560→          state.turn,
   561→          "system",
   562→          `¡Trampa activada en ${lane}! -${activeTrap?.value ?? 0} al poder.`,
   563→          "debuff"
   564→        );
   565→      }
   566→      if (res.newTrap) {
   567→        log = pushLog(
   568→          log,
   569→          state.turn,
   570→          "player",
   571→          `Trampa colocada en ${lane}.`,
   572→          "info"
   573→        );
   574→      }
   575→
   576→      // ====== TRIGGER DISPATCH ======
   577→      // ON_DEATH: for each card destroyed by this play, fire its on_death effect.
   578→      // The dying card's full state is recovered from the pre-mutation board.
   579→      let triggerBoard = res.board;
   580→      for (const rc of res.removedCards) {
   581→        const deadCard = state.board[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   582→        if (!deadCard) continue;
   583→        const deathRes = applyOnDeathEffects(triggerBoard, rc.side, rc.lane, deadCard);
   584→        triggerBoard = deathRes.board;
   585→        log = pushTriggerLogs(log, state.turn, deathRes);
   586→      }
   587→
   588→      // ====== POWER LOG TRACKING ======
   589→      const playedUid = res.playedCard.uid;
   590→      const _playedOnBoard = triggerBoard[lane]["player" as "player" | "enemy"].find(c => c.uid === playedUid)
   591→        ?? triggerBoard[lane]["enemy" as "player" | "enemy"].find(c => c.uid === playedUid);
   592→      if (_playedOnBoard) {
   593→        const pLog: PowerLogEntry[] = [];
   594→        if (card.handBonus) pLog.push({ source: card.keyword ? KEYWORD_META[card.keyword].label : "Bonificador en mano", delta: card.handBonus, kind: "hand" });
   595→        if (card.subtypes.includes(lane)) pLog.push({ source: "Sinergia", delta: 2, kind: "synergy" });
   596→        const lbGain = getLaneBonusPower(card.category, laneBonus);
   597→        if (lbGain > 0) pLog.push({ source: laneBonus?.label ?? "Campo", delta: lbGain, kind: "field" });
   598→        const effBonus = res.playedCard.currentPower - card.power - card.handBonus - (card.subtypes.includes(lane) ? 2 : 0) - lbGain;
   599→        if (effBonus !== 0) pLog.push({ source: "Efecto propio", delta: effBonus, kind: "effect" });
   600→        if (res.trapTriggered && activeTrap) pLog.push({ source: "Trampa enemiga", delta: -activeTrap.value, kind: "trap" });
   601→        _playedOnBoard.powerLog = pLog;
   602→      }
   603→      for (const m of res.mutations) {
   604→        const arr = triggerBoard[m.lane][m.side];
   605→        const tgt = arr.find(c => c.uid === m.uid);
   606→        if (tgt) {
   607→          if (!tgt.powerLog) tgt.powerLog = [];
   608→          tgt.powerLog.push({ source: card.name, delta: m.powerDelta, kind: m.powerDelta > 0 ? "buff" : "debuff" });
   609→        }
   610→      }
   611→
   612→      set({
   613→        board: triggerBoard,
   614→        hand: res.hand,
   615→        energy: res.remainingEnergy,
   616→        laneTraps: newLaneTraps,
   617→        nextTurnEnergyPenalty,
   618→        floatingNumbers,
   619→        gameStats: stats,
   620→        undoSnapshot,
   621→        playsThisTurn: state.playsThisTurn + 1,
   622→        legendaryBurst: card.rarity === "legendary",
   623→        selectedCardUid: null,
   624→        invalidPlayMsg: null,
   625→        log,
   626→      });
   627→
   628→      // Clear floating numbers after they've had time to animate
   629→      setTimeout(() => {
   630→        get().clearFloatingNumbers();
   631→      }, 1500);
   632→
   633→      // Clear legendary burst after the animation completes
   634→      if (card.rarity === "legendary") {
   635→        setTimeout(() => {
   636→          set({ legendaryBurst: false });
   637→        }, 1900);
   638→      }
   639→    } catch (e) {
   640→      set({
   641→        invalidPlayMsg: (e as Error).message,
   642→      });
   643→    }
   644→  },

   646→  endTurn: async () => {
   647→    const state = get();
   648→    if (state.phase !== "playing") return;

   650→    // Pasar button click sound

   681→
   682→    // 2. Build the enemy hand: draw 3 random cards from a fresh shuffled pile.

   684→    const enemyEnergy = freshState.maxEnergy;

   697→      enemyHand,
   698→      enemyEnergy,
   699→      freshState.laneBonuses,

   703→    let workEnemyHand = [...enemyHand];

   714→          workEnemyHand,

   716→          "enemy",

   718→          play.card,

   723→        workEnemyHand = res.hand;

   726→        // Update traps: clear if triggered, set if newTrap

   738→          if (res.mutations.some((m) => m.powerDelta < 0)) sfx.debuff();

   745→        gameStats.buffsApplied += res.mutations.filter(

   748→        // debuffs to PLAYER cards count as "debuffs to enemy" from enemy's
   749→        // perspective — we still keep a single "debuffsToEnemy" counter that
   750→        // tracks debuffs applied TO the enemy side, so for enemy plays we
   751→        // don't increment it (the enemy isn't debuffing itself).
   752→        // (Player cards destroyed by enemy plays don't increment
   753→        // `enemyCardsDestroyed` either — that stat is player-centric.)

   765→            log = pushLog(

   770→              "buff"
   771→            );
   772→            continue;
   773→          }
   774→          const sign = m.powerDelta > 0 ? "+" : "";
   775→          log = pushLog(
   776→            log,
   777→            freshState.turn,
   778→            m.side === "player" ? "player" : "enemy",

   780→            m.powerDelta > 0 ? "buff" : "debuff"
   781→          );
   782→        }

   798→            "debuff"

   800→        }

   802→          log = pushLog(
   803→            log,
   804→            freshState.turn,
   805→            "enemy",
   806→            `Trampa enemiga colocada en ${play.lane}.`,
   807→            "info"
   808→          );
   809→        }

   811→        // ====== TRIGGER DISPATCH (enemy play) ======
   812→        // ON_DEATH: for each card destroyed by the enemy's play.
   813→        // Look up the dying card in the PRE-play board (prePlayBoard) because
   814→        // workBoard (= res.board) has already had removedCards filtered out.
   815→        for (const rc of res.removedCards) {
   816→          const deadCard = prePlayBoard[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   817→          if (!deadCard) continue;
   818→          const deathRes = applyOnDeathEffects(workBoard, rc.side, rc.lane, deadCard);
   819→          workBoard = deathRes.board;
   820→          log = pushTriggerLogs(log, freshState.turn, deathRes);
   821→        }
   822→      } catch {
   823→        // ignore failed plays
   824→      }
   825→    }
   826→
   827→    // ====== TRIGGER: END_TURN (enemy) ======
   828→    // Enemy just finished playing — fire their end_turn effects before turn advances.

   830→    workBoard = enemyEndRes.board;
   831→    log = pushTriggerLogs(log, freshState.turn, enemyEndRes);
   832→

   834→    const newTurn = freshState.turn + 1;
   835→    const newMaxEnergy = Math.min(MAX_TURNS, freshState.maxEnergy + 1);
   836→    const newEnergy = Math.max(0, newMaxEnergy + freshState.nextTurnEnergyPenalty);
   837→
   838→    if (newTurn > MAX_TURNS) {
   839→      // Game over
   840→      const finalResult = computeFinalResult(workBoard);
   841→      log = pushLog(
   842→        log,
   843→        MAX_TURNS,
   844→        "system",
   845→        `Fin del juego: ${finalResult.result === "WIN" ? "¡Victoria!" : finalResult.result === "LOSS" ? "Derrota" : "Empate"} (${finalResult.playerWins}-${finalResult.enemyWins}, E ${finalResult.draws}).`,
   846→        "end"

   857→        lastEnemyPlays: plan,
   858→        laneTraps: workTraps,
   859→        nextTurnEnergyPenalty: 0,

   861→        log,
   862→        gameStats,
   863→        playsThisTurn: 0,
   864→      });

   866→    }

   870→    let newHand = [...freshState.hand];
   871→    if (newHand.length < MAX_HAND_SIZE && newDrawPile.length > 0) {
   872→      const drawn = newDrawPile.pop()!;
   873→      newHand.push({ ...drawn, uid: uuidv4(), handBonus: 0 });
   874→    }
   875→
   876→    log = pushLog(
   877→      log,
   878→      newTurn,
   879→      "system",
   880→      `Turno ${newTurn} — Energía ${newEnergy}/${newMaxEnergy}${freshState.nextTurnEnergyPenalty ? ` (penalización ${freshState.nextTurnEnergyPenalty})` : ""}.`,
   881→      "info"
   882→    );
   883→
   884→    // ====== TRIGGER: START_TURN (player) ======
   885→    // Player's new turn begins — fire their start_turn effects before they play.
   886→    const playerStartRes = applyStartTurnEffects(workBoard, "player", newHand);
   887→    workBoard = playerStartRes.board;
   888→    log = pushTriggerLogs(log, newTurn, playerStartRes);
   889→
   890→    // ====== RE-ROLL LANE BONUSES AT TURN 4 ======
   891→    // User spec: at the start of turn 4, do another random roll of lane bonuses
   892→    // so the player needs to adapt their strategy mid-match (was static before).

   894→    if (newTurn === 4) {
   895→      newLaneBonuses = rollLaneBonuses();
   896→      const bonusSummary = LANE_IDS.map((ln) => {
   897→        const b = newLaneBonuses[ln];
   898→        return b ? `${ln}: ${b.label}` : `${ln}: sin bono`;
   899→      }).join(" · ");
   900→      log = pushLog(log, newTurn, "system", `🔮 ¡Bonos re-rollados! ${bonusSummary}`, "buff");
   901→      // Trigger visual flash on each lane showing the new bonus
   902→      flashLaneBonuses(newLaneBonuses);
   903→    }
   904→
   905→    set({
   906→      board: workBoard,
   907→      turn: newTurn,
   908→      energy: newEnergy,
   909→      maxEnergy: newMaxEnergy,
   910→      drawPile: newDrawPile,
   911→      hand: newHand,
   912→      phase: "playing",
   913→      selectedCardUid: null,
   914→      invalidPlayMsg: null,
   915→      lastEnemyPlays: plan,
   916→      laneTraps: workTraps,
   917→      laneBonuses: newLaneBonuses,
   918→      nextTurnEnergyPenalty: 0,
   919→      aiTurnPhase: "idle",
   920→      log,

   992→  swapCardInDeck: (index, cardId) => {
   993→    // Allow placing a card into ANY of the 12 deck slots, even if the slot is
   994→    // beyond the current deckCardIds length (e.g. user emptied the deck then
   995→    // clicks slot 5 — we pad with empty strings up to that index).
   996→    if (index < 0 || index >= 12) return;
   997→    const { deckCardIds } = get();
   998→    if (deckCardIds[index] === cardId) return;
   999→    // Prevent duplicates (deck is 12 unique cards) — if the card is already in
  1000→    // the deck at a DIFFERENT slot, move it from the old slot to the new one
  1001→    // (old slot becomes empty). This is the "drag-to-rearrange" UX.
  1002→    const next = [...deckCardIds];
  1003→    // Pad with empty strings up to the target index so we can place at any slot
  1004→    while (next.length <= index) next.push("");
  1005→    if (deckCardIds.includes(cardId)) {
  1006→      const existingIdx = deckCardIds.indexOf(cardId);
  1007→      next[existingIdx] = "";  // clear old slot
  1008→    }
  1009→    next[index] = cardId;
  1010→    // Always store exactly 12 entries (pad/truncate to 12)
  1011→    while (next.length < 12) next.push("");
  1012→    set({ deckCardIds: next.slice(0, 12), deckDirty: true, deckSaveStatus: "idle" });
  1013→  },
  1014→
  1015→  saveDeck: async () => {
  1016→    const { deckCardIds } = get();
  1017→    if (deckCardIds.length !== 12) return;
  1018→    set({ deckSaveStatus: "saving" });
  1019→    try {
  1020→      // Save to localStorage (no API route needed — works in static export / APK)
  1021→      localStorage.setItem("guerrero-tactico-deck", JSON.stringify({ cardIds: deckCardIds }));
  1022→      set({ deckSaveStatus: "ok", deckDirty: false });
  1023→      setTimeout(() => {
  1024→        if (get().deckSaveStatus === "ok") set({ deckSaveStatus: "idle" });
  1025→      }, 2000);
  1026→    } catch (e) {
  1027→      console.error("saveDeck error:", e);
  1028→      set({ deckSaveStatus: "err" });
  1029→    }
  1030→  },
  1031→
  1032→  resetDeck: async () => {

  1046→// Convenience selector for lane scores
  1047→export function laneScores(board: Board, lane: LaneId) {
  1048→  const p = board[lane].player.reduce((a, c) => a + c.currentPower, 0);
  1049→  const e = board[lane].enemy.reduce((a, c) => a + c.currentPower, 0);
  1050→  return { player: p, enemy: e };
  1051→}
