  1660→  // ====== BUFFS (delta >= 0) — bypass pipeline ======
  1661→  if (delta >= 0) {
  1662→    applyPowerDelta(board, side, lane, uid, delta, source, kind);
  1663→    return {
  1664→      appliedDelta: delta,
  1665→      absorbedByCupula: 0,
  1666→      finalTargetUid: uid,
  1667→      finalLane: lane,
  1668→      blockedByBaluarte: false,
  1669→    };
  1670→  }
  1671→
  1672→  // ====== DEBUFF PIPELINE (delta < 0) ======
  1673→  const arr = board[lane][side];
  1674→  const idx = arr.findIndex((c) => c.uid === uid);
  1675→  if (idx < 0) {
  1676→    return {
  1677→      appliedDelta: 0,
  1678→      absorbedByCupula: 0,
  1679→      finalTargetUid: uid,
  1680→      finalLane: lane,
  1681→      blockedByBaluarte: false,
  1682→    };
  1683→  }
  1684→  const target = arr[idx];
  1685→  const sourceSide = ctx?.sourceSide ?? side;
  1686→  const isEnemyDebuff = sourceSide !== side;
  1687→  const mag = Math.abs(delta);
  1688→
  1689→  // ----- STEP 1: GUARDIÁN redirect -----
  1690→  // If any ally in this lane (other than the target) is a Guardián, redirect the
  1691→  // debuff to it. The Guardián then runs through the rest of the pipeline
  1692→  // (Baluarte / Cúpula / applyPowerDelta / reactions) on itself.
  1693→  if (!target.isGuardian) {
  1694→    const guardian = arr.find((c) => c.uid !== uid && c.isGuardian);
  1695→    if (guardian) {
  1696→      const sub = applyDebuff(board, side, lane, guardian.uid, delta, source, kind, ctx);
  1697→      return {
  1698→        ...sub,
  1699→        redirectedByGuardian: guardian.uid,
  1700→      };
  1701→    }
  1702→  }
  1703→
  1704→  // ----- STEP 2: BALUARTE immunity (dynamic check) -----
  1705→  // If any ally in this lane (other than the target itself) has keyword "baluarte",
  1706→  // the target is immune to enemy debuffs. Perforación leaks up to `bypass` points
  1707→  // through this immunity (Baluarte is modeled as "infinite Cúpula").
  1708→  let effectiveDelta = delta; // negative
  1709→  let absorbedByCupula = 0;
  1710→  let blockedByBaluarte = false;
  1711→  let perforacionBypass = ctx?.perforacionBypass ?? 0;
  1712→
  1713→  // Pull perforacionRemaining from the source card if not explicitly provided.
  1714→  if (ctx?.sourceUid && ctx.sourceSide !== undefined && perforacionBypass === 0) {
  1715→    const sArr = board[ctx.sourceLane ?? lane]?.[ctx.sourceSide];
  1716→    const sCard = sArr?.find((c) => c.uid === ctx.sourceUid);
  1717→    if (sCard?.perforacionRemaining) {
  1718→      perforacionBypass = sCard.perforacionRemaining;
  1719→    }
  1720→  }
  1721→
  1722→  const hasBaluarte = arr.some((c) => c.uid !== target.uid && c.keyword === "baluarte");
  1723→  if (hasBaluarte && isEnemyDebuff) {
  1724→    // Baluarte: infinite absorption, but Perforación leaks up to `bypass` points.
  1725→    const leak = Math.min(mag, perforacionBypass);
  1726→    if (leak < mag) {
  1727→      // Blocked entirely (or partially — but Baluarte is "infinite", so any
  1728→      // un-leaked amount is fully blocked).
  1729→      blockedByBaluarte = true;
  1730→      // Still log the leak as a residual debuff.
  1731→      effectiveDelta = leak > 0 ? -leak : 0;
  1732→      absorbedByCupula = mag - leak; // counts as absorbed by Baluarte
  1733→      // Consume perforacionRemaining on the source if it leaked.
  1734→      if (leak > 0 && ctx?.sourceUid && ctx.sourceSide !== undefined) {
  1735→        const sArr2 = board[ctx.sourceLane ?? lane]?.[ctx.sourceSide];
  1736→        const sIdx = sArr2?.findIndex((c) => c.uid === ctx.sourceUid);
  1737→        if (sArr2 && sIdx !== undefined && sIdx >= 0) {
  1738→          sArr2[sIdx] = {
  1739→            ...sArr2[sIdx],
  1740→            perforacionRemaining: Math.max(0, (sArr2[sIdx].perforacionRemaining ?? 0) - leak),
  1741→          };
  1742→        }
  1743→      }
  1744→    } else {
  1745→      // Perforación fully bypasses Baluarte — apply the full debuff.
  1746→      effectiveDelta = delta;
  1747→      // Consume all perforacionRemaining on the source.
  1748→      if (ctx?.sourceUid && ctx.sourceSide !== undefined) {
  1749→        const sArr2 = board[ctx.sourceLane ?? lane]?.[ctx.sourceSide];
  1750→        const sIdx = sArr2?.findIndex((c) => c.uid === ctx.sourceUid);
  1751→        if (sArr2 && sIdx !== undefined && sIdx >= 0) {
  1752→          sArr2[sIdx] = {
  1753→            ...sArr2[sIdx],
  1754→            perforacionRemaining: 0,
  1755→          };
  1756→        }
  1757→      }
  1758→    }
  1759→  } else {
  1760→    // ----- STEP 3: CÚPULA absorption -----
  1761→    // Each ally in the lane with `cupulaRemaining > 0` absorbs points from the
  1762→    // debuff. Perforación reduces the absorption capacity: for a Cúpula with
  1763→    // pool P, the effective capacity is max(0, P - perforacionBypass).
  1764→    //
  1765→    // We absorb from the Cúpula with the largest remaining pool first (greedy).
  1766→    // Each absorbed point decrements that Cúpula's pool.
  1767→    // Perforación bypass is consumed across all Cúpula cards it touches.
  1768→    let remaining = mag;
  1769→    let bypassLeft = perforacionBypass;
  1770→
  1771→    // Collect Cúpula cards in this lane (target included — Cúpula is PROPIO+CARRIL_LOCAL per spec).
  1772→    const cupulaCards = arr
  1773→      .filter((c) => (c.cupulaRemaining ?? 0) > 0)
  1774→      .sort((a, b) => (b.cupulaRemaining ?? 0) - (a.cupulaRemaining ?? 0));
  1775→
  1776→    for (const cup of cupulaCards) {
  1777→      if (remaining <= 0) break;
  1778→      const cupIdx = arr.findIndex((c) => c.uid === cup.uid);
  1779→      if (cupIdx < 0) continue;
  1780→      const pool = cup.cupulaRemaining ?? 0;
  1781→      // Perforación reduces this Cúpula's effective capacity.
  1782→      const bypassHere = Math.min(bypassLeft, pool);
  1783→      const effectiveCapacity = Math.max(0, pool - bypassHere);
  1784→      const absorb = Math.min(remaining, effectiveCapacity);
  1785→      if (absorb > 0) {
  1786→        arr[cupIdx] = {
  1787→          ...cup,
  1788→          cupulaRemaining: Math.max(0, pool - absorb),
  1789→        };
  1790→        absorbedByCupula += absorb;
  1791→        remaining -= absorb;
  1792→      }
  1793→      // Consume the bypass used here (regardless of whether absorb happened).
  1794→      bypassLeft -= bypassHere;
  1795→    }
  1796→
  1797→    // ----- STEP 3b: REFUGIO absorption -----
  1798→    // Allies with keyword=refugio absorb up to keywordValue remaining debuff points
  1799→    // directed at OTHER allies in this lane (not the Refugio card itself).
  1800→    // Uses keywordValue as a one-shot pool (decrements when used, not refreshed).
  1801→    if (remaining > 0) {
  1802→      const refugioCards = arr
  1803→        .filter((c) => c.uid !== target.uid && c.keyword === "refugio" && (c.keywordValue ?? 0) > 0)
  1804→        .sort((a, b) => (b.keywordValue ?? 0) - (a.keywordValue ?? 0));
  1805→      for (const ref of refugioCards) {
  1806→        if (remaining <= 0) break;
  1807→        const refIdx = arr.findIndex((c) => c.uid === ref.uid);
  1808→        if (refIdx < 0) continue;
  1809→        const pool = ref.keywordValue ?? 0;
  1810→        const absorb = Math.min(remaining, pool);
  1811→        if (absorb > 0) {
  1812→          arr[refIdx] = {
  1813→            ...ref,
  1814→            keywordValue: Math.max(0, pool - absorb),
  1815→            powerLog: [
  1816→              ...(ref.powerLog ?? []),
  1817→              { source: `${ref.name} (Refugio)`, delta: absorb, kind: "passive" },
  1818→            ],
  1819→          };
  1820→          absorbedByCupula += absorb;
  1821→          remaining -= absorb;
  1822→        }
  1823→      }
  1824→    }
  1825→
  1826→    effectiveDelta = -remaining;
  1827→    // Consume perforacionRemaining on the source proportional to bypassLeft used.
  1828→    const bypassUsed = perforacionBypass - bypassLeft;
  1829→    if (bypassUsed > 0 && ctx?.sourceUid && ctx.sourceSide !== undefined) {
  1830→      const sArr2 = board[ctx.sourceLane ?? lane]?.[ctx.sourceSide];
  1831→      const sIdx = sArr2?.findIndex((c) => c.uid === ctx.sourceUid);
  1832→      if (sArr2 && sIdx !== undefined && sIdx >= 0) {
  1833→        sArr2[sIdx] = {
  1834→          ...sArr2[sIdx],
  1835→          perforacionRemaining: Math.max(0, (sArr2[sIdx].perforacionRemaining ?? 0) - bypassUsed),
  1836→        };
  1837→      }
  1838→    }
  1839→  }
  1840→
  1841→  // ----- STEP 4: apply residual delta -----
  1842→  if (effectiveDelta !== 0) {
  1843→    applyPowerDelta(board, side, lane, target.uid, effectiveDelta, source, kind);
  1844→  }
  1845→
  1846→  // ----- STEP 5: REACTIONS via event system (Etapa 4) -----
  1847→  // Espinas/Enrabia used to be inline here; they're now dispatched by
  1848→  // `applyReactionEffects` after the engine emits a `debuff_applied` event.
  1849→  // Only fires if the target actually took damage (effectiveDelta < 0) and the
  1850→  // source is identifiable (ctx.sourceUid known — on_death debuffs have no
  1851→  // source on the board, so they skip reactions but Cúpula/Guardián/Baluarte
  1852→  // still applied in Steps 2-3).
  1853→  let espinasFired = false;
  1854→  let enrabiaFired = false;
  1855→
  1856→  if (effectiveDelta < 0 && isEnemyDebuff && ctx?.sourceUid && ctx.sourceSide !== undefined) {
  1857→    const reactionEvent: DebuffAppliedEvent = {
  1858→      type: "debuff_applied",
  1859→      sourceSide: ctx.sourceSide,
  1860→      sourceUid: ctx.sourceUid,
  1861→      sourceName: source,
  1862→      sourceLane: ctx.sourceLane ?? lane,
  1863→      targetSide: side,
  1864→      targetUid: target.uid,
  1865→      targetName: target.name,
  1866→      targetLane: lane,
  1867→      delta: effectiveDelta,
  1868→    };
  1869→    const rxnRes = applyReactionEffects(board, side, reactionEvent);
  1870→    espinasFired = rxnRes.espinasFired;
  1871→    enrabiaFired = rxnRes.enrabiaFired;
  1872→  }
  1873→
  1874→  return {
  1875→    appliedDelta: effectiveDelta,
  1876→    absorbedByCupula,
  1877→    finalTargetUid: target.uid,
  1878→    finalLane: lane,
  1879→    blockedByBaluarte,
  1880→    espinasFired,
  1881→    enrabiaFired,
  1882→  };
  1883→}
  1884→
  1885→// ===========================================================================
  1886→// ETAPA 4 — REACTION TRIGGER EVENT SYSTEM (spec 2026-10)
  1887→// ===========================================================================
  1888→// Top-level game events that REACTION-trigger cards can respond to. The engine
  1889→// emits these events; `applyReactionEffects` dispatches them to all reaction
  1890→// cards on the reacting side.
  1891→//
  1892→// Currently only Espinas and Enrabia have REACTION trigger, and both respond
  1893→// to `debuff_applied` events. Future REACTION keywords can react to
  1894→// `card_played` or `card_destroyed` events.
  1895→// ===========================================================================
  1896→
  1897→/** Event emitted when a card is played to the board. */
  1898→export interface CardPlayedEvent {

  1903→  cardUid: string;
  1904→  cardName: string;
  1905→  cardCategory: CardCategory;
  1906→  cardSubtypes: Subtype[];
  1907→}
  1908→

  1912→  /** Side of the source card that applied the debuff. */
  1913→  sourceSide: "player" | "enemy";
  1914→  sourceUid: string;
  1915→  sourceName: string;
  1916→  sourceLane: LaneId;
  1917→  /** Side of the target card that received the debuff. */
  1918→  targetSide: "player" | "enemy";
  1919→  targetUid: string;
  1920→  targetName: string;

  1923→  delta: number;

  1925→

  1938→
  1939→/** Union of all top-level game events REACTION cards can respond to. */
  1940→export type GameEvent = CardPlayedEvent | DebuffAppliedEvent | CardDestroyedEvent;
  1941→
  1942→export interface ReactionResult {
  1943→  espinasFired?: boolean;
  1944→  enrabiaFired?: boolean;
  1945→  logs: TriggerResult["logs"];
  1946→}
  1947→
  1948→/**
  1949→ * Dispatch a game event to all REACTION-trigger cards on `reactingSide`.
  1950→ * Iterates over every card on the reacting side with `trigger === "reaction"`
  1951→ * and invokes the keyword-specific handler if the event matches.
  1952→ *
  1953→ * Returns logs + flags indicating which reactions fired (useful for tests and
  1954→ * for the game log).
  1955→ *
  1956→ * NOTE: Reactions that apply debuffs (e.g. Espinas counter-attack) use
  1957→ * `applyPowerDelta` directly — NOT `applyDebuff` — to avoid infinite recursion
  1958→ * (a counter-debuff would otherwise emit another `debuff_applied` event).
  1959→ */
  1960→export function applyReactionEffects(
  1961→  board: Board,
  1962→  reactingSide: "player" | "enemy",
  1963→  event: GameEvent,
  1964→): ReactionResult {
  1965→  const logs: TriggerResult["logs"] = [];
  1966→  let espinasFired = false;
  1967→  let enrabiaFired = false;
  1968→
  1969→  for (const lane of LANE_IDS) {
  1970→    const arr = board[lane][reactingSide];
  1971→    for (let i = 0; i < arr.length; i++) {
  1972→      const card = arr[i];
  1973→      if (card.trigger !== "reaction") continue;
  1974→
  1975→      // ----- ESPINAS [X]: reacts to debuff_applied where target is self -----
  1976→      if (
  1977→        card.keyword === "espinas" &&
  1978→        card.keywordValue &&
  1979→        event.type === "debuff_applied" &&
  1980→        event.targetSide === reactingSide &&
  1981→        event.targetUid === card.uid
  1982→      ) {
  1983→        const x = card.keywordValue;
  1984→        // Counter-attack: -X to the source card (applyPowerDelta, NOT applyDebuff,
  1985→        // to avoid infinite recursion).
  1986→        const sArr = board[event.sourceLane]?.[event.sourceSide];
  1987→        const sIdx = sArr?.findIndex((c) => c.uid === event.sourceUid);
  1988→        if (sArr && sIdx !== undefined && sIdx >= 0) {
  1989→          applyPowerDelta(
  1990→            board,
  1991→            event.sourceSide,
  1992→            event.sourceLane,
  1993→            event.sourceUid,
  1994→            -x,
  1995→            `${card.name} (Espinas)`,
  1996→            "debuff",
  1997→          );
  1998→          espinasFired = true;
  1999→          logs.push({
  2000→            side: reactingSide,
  2001→            text: `🌵 ${card.name} (Espinas): -${x} a ${event.sourceName} (contraataque)`,
  2002→          });
  2003→        }
  2004→      }
  2005→
  2006→      // ----- ENRABIA [X]: one-shot +X to self when debuffed -----
  2007→      if (
  2008→        card.keyword === "enrabia" &&
  2009→        card.keywordValue &&
  2010→        !card.enrabiadUsed &&
  2011→        event.type === "debuff_applied" &&
  2012→        event.targetSide === reactingSide &&
  2013→        event.targetUid === card.uid
  2014→      ) {
  2015→        const x = card.keywordValue;
  2016→        // Re-fetch the card from the array (applyPowerDelta may have replaced
  2017→        // the array element via spread).
  2018→        const tIdx = board[lane][reactingSide].findIndex((c) => c.uid === card.uid);
  2019→        if (tIdx >= 0) {
  2020→          board[lane][reactingSide][tIdx] = {
  2021→            ...board[lane][reactingSide][tIdx],
  2022→            enrabiadUsed: true,
  2023→          };
  2024→          applyPowerDelta(
  2025→            board,
  2026→            reactingSide,
  2027→            lane,
  2028→            card.uid,
  2029→            x,
  2030→            `${card.name} (Enrabia)`,
  2031→            "buff",
  2032→          );
  2033→          enrabiaFired = true;
  2034→          logs.push({
  2035→            side: reactingSide,
  2036→            text: `😡 ${card.name} (Enrabia): +${x} (primera reducción sufrida)`,
  2037→          });
  2038→        }
  2039→      }
  2040→
  2041→      // ----- Future REACTION keywords can hook here -----
  2042→      // e.g. a "Centinela" keyword reacting to `card_played` events where
  2043→      // event.side !== reactingSide (rival played a card) and event.cardCategory
  2044→      // matches some condition.
  2045→    }
  2046→  }
  2047→
  2048→  return { espinasFired, enrabiaFired, logs };
  2049→}
  2050→
  2051→/**
  2052→ * ETAPA 3 — Frenesí [X] helper.
  2053→ * Frenesí grants +X to the card itself every time its trigger fires. This helper
  2054→ * is invoked from EVERY trigger dispatcher (start_turn / end_turn / on_death /
  2055→ * umbral) AFTER the main effect runs, so the cumulative +X composes with the
  2056→ * card's normal effect.
  2057→ *
  2058→ * On ON_DEATH the helper is a no-op (the card is leaving the board, so a +X
  2059→ * bonus would be wasted).
  2060→ *
  2061→ * The +X is logged via `logs` so the game log shows the Frenesí stack.
  2062→ */
  2063→function triggerFrenesiIfApplicable(
  2064→  board: Board,
  2065→  side: "player" | "enemy",
  2066→  lane: LaneId,
  2067→  cardUid: string,
  2068→  cardName: string,
  2069→  triggerLabel: string,
  2070→  logs: TriggerResult["logs"],
  2071→): void {
  2072→  const arr = board[lane][side];
  2073→  const idx = arr.findIndex((c) => c.uid === cardUid);
  2074→  if (idx < 0) return;
  2075→  const card = arr[idx];
  2076→  if (card.keyword !== "frenesi" || !card.keywordValue) return;
  2077→  const x = card.keywordValue;
  2078→  applyPowerDelta(board, side, lane, cardUid, x, `${cardName} (Frenesí)`, "buff");
  2079→  logs.push({
  2080→    side,
  2081→    text: `🌀 ${cardName} (Frenesí): +${x} por disparar ${triggerLabel}`,
  2082→  });
  2083→}
  2084→
  2085→// ===========================================================================
  2086→// FIRE KEYWORD EFFECTS — dispatcher genérico (spec 2026-10 amendment)
  2087→// ===========================================================================
  2088→// Llamado desde TODOS los dispatchers de trigger (on_play, start_turn, end_turn,
  2089→// despertar, y el caso Sacrificio para ofrenda). Esto permite que CUALQUIER
  2090→// keyword dispare con CUALQUIER trigger sin causar confusión.
  2091→//
  2092→// Solo maneja keywords que tienen sentido como EFECTOS one-shot o per-turn
  2093→// (no passive auras como Baluarte/Sigilo/Cúpula/Guardián que se chequean
  2094→// dinámicamente, ni keywords que requieren removedCards/energy/deck como
  2095→// Sacrificio/Remate/Reagrupar/Trampa/Retorno/Reciclar/Renovación).
  2096→// ===========================================================================
  2097→
  2098→function fireKeywordEffects(
  2099→  board: Board,
  2100→  side: "player" | "enemy",
  2101→  lane: LaneId,
  2102→  card: BoardCard,
  2103→  hand: HandCard[] = [],
  2104→  logs: TriggerResult["logs"] = [],
  2105→  /** When true (called from start_turn/end_turn dispatchers), skip conditional
  2106→   *  self-buff keywords (Ventaja, Aislado, Emboscada, Vanguardia, Mimetismo)
  2107→   *  that would stack indefinitely if fired every turn. These should only fire
  2108→   *  once at on_play or despertar (recurring=false). */
  2109→  recurring = false,
  2110→): void {
  2111→  if (!card.keyword || card.keyword === "none") return;
  2112→  const x = card.keywordValue ?? 0;
  2113→  const enemySide: "player" | "enemy" = side === "player" ? "enemy" : "player";
  2114→
  2115→  // Skip conditional self-buff keywords when recurring (start_turn/end_turn).
  2116→  // These would stack +X every turn if allowed to fire repeatedly.
  2117→  if (recurring && (
  2118→    card.keyword === "ventaja" ||
  2119→    card.keyword === "aislado" ||
  2120→    card.keyword === "emboscada" ||
  2121→    card.keyword === "vanguardia" ||
  2122→    card.keyword === "mimetismo"
  2123→  )) {
  2124→    return;
  2125→  }
  2126→
  2127→  switch (card.keyword) {
  2128→    // ---- SACRIFICIO: +X a sí misma (el bonus por sacrificar un aliado) ----
  2129→    case "sacrificio": {
  2130→      if (x > 0) {
  2131→        applyPowerDelta(board, side, lane, card.uid, x, `${card.name} (Sacrificio)`, "buff");
  2132→        logs.push({ side, text: `🗡️ ${card.name} (Sacrificio): +${x} a sí misma` });
  2133→      }
  2134→      break;
  2135→    }
  2136→
  2137→    // ---- REMATE: destruir enemigo con poder ≤ X ----
  2138→    case "remate": {
  2139→      if (x > 0) {
  2140→        const target = highestPowerCard(board[lane][enemySide]);
  2141→        if (target && target.currentPower <= x) {
  2142→          // Mark for removal — but fireKeywordEffects doesn't have removedCards access.
  2143→          // For now, just apply -target.currentPower to "destroy" it (set to 0).
  2144→          // The gameStore should handle actual removal separately.
  2145→          applyPowerDelta(board, enemySide, lane, target.uid, -target.currentPower, `${card.name} (Remate)`, "debuff");
  2146→          logs.push({ side, text: `💀 ${card.name} (Remate): destruyó ${target.name} (poder ${target.currentPower} ≤ ${x})` });
  2147→        }
  2148→      }
  2149→      break;
  2150→    }
  2151→
  2152→    // ---- CELERIDAD: reduce coste de carta ALEATORIA en mano ----
  2153→    case "celeridad": {
  2154→      if (x > 0 && hand.length > 0) {
  2155→        const randomIdx = Math.floor(Math.random() * hand.length);
