  2692→        });
  2693→        if (res.blockedByBaluarte || res.appliedDelta === 0) {
  2694→          logs.push({
  2695→            side,
  2696→            text: `💀 ${deadCard.name}: -3 a ${target.name} fue BLOQUEADO (Baluarte) en carril ${lane}`,
  2697→          });
  2698→        } else {
  2699→          logs.push({
  2700→            side,
  2701→            text: `💀 ${deadCard.name}: ${res.appliedDelta} a ${target.name} en carril ${lane} (Último Aliento, post-prevención)`,
  2702→          });
  2703→        }
  2704→      }
  2705→      break;
  2706→    }
  2707→    case "filotormenta": {
  2708→      // -1 to ALL enemies on board on death (each debuff goes through the pipeline)
  2709→      let debuffed = 0;
  2710→      let blocked = 0;
  2711→      for (const ln of LANE_IDS) {
  2712→        // Snapshot the UIDs first since applyDebuff may mutate the array.
  2713→        const enemyUids = newBoard[ln][enemySide].map((c) => c.uid);
  2714→        for (const eUid of enemyUids) {
  2715→          const res = applyDebuff(newBoard, enemySide, ln, eUid, -1, deadCard.name, "on_death", {});
  2716→          if (res.blockedByBaluarte || res.appliedDelta === 0) blocked++;
  2717→          else debuffed++;
  2718→        }
  2719→      }
  2720→      if (debuffed > 0 || blocked > 0) {
  2721→        logs.push({
  2722→          side,
  2723→          text: `💀 ${deadCard.name}: -1 a ${debuffed} enemigo(s) en mesa${blocked > 0 ? `, ${blocked} bloqueado(s) por Baluarte` : ""} (Último Aliento)`,
  2724→        });
  2725→      }
  2726→      break;
  2727→    }
  2728→    case "ultimo_aliento": {
  2729→      // 13 Último Aliento [X]: On Death - transfer +X to weakest ally in this lane
  2730→      const x = deadCard.keywordValue ?? 3;
  2731→      const target = lowestPowerCard(newBoard[lane][side]);
  2732→      if (target) {
  2733→        applyPowerDelta(newBoard, side, lane, target.uid, x, deadCard.name, "on_death");
  2734→        logs.push({
  2735→          side,
  2736→          text: `💀 ${deadCard.name}: +${x} a ${target.name} (Último Aliento)`,
  2737→        });
  2738→      }
  2739→      break;
  2740→    }
  2741→    case "reciclar": {
  2742→      // 20 Reciclar [X]: On Death - recover X energy.
  2743→      // gameStore handles the actual energy refund (we can't `set` energy from
  2744→      // the engine). Log the intent.
  2745→      const x = deadCard.keywordValue ?? 2;
  2746→      logs.push({
  2747→        side,
  2748→        text: `♻️ ${deadCard.name}: recupera ${x} de energía (Reciclar)`,
  2749→      });
  2750→      break;
  2751→    }
  2752→    case "retorno": {
  2753→      // 18 Retorno [X]: On Death - if cost ≤ X, return to hand.
  2754→      // gameStore handles the actual hand insertion. Log the intent.
  2755→      const x = deadCard.keywordValue ?? deadCard.cost;
  2756→      if (deadCard.cost <= x) {
  2757→        logs.push({
  2758→          side,
  2759→          text: `🔄 ${deadCard.name} regresa a la mano (Retorno: coste ${deadCard.cost} ≤ ${x})`,
  2760→        });
  2761→      }
  2762→      break;
  2763→    }
  2764→    default:
  2765→      break;
  2766→  }
  2767→
  2768→  return { board: newBoard, logs, umbralFired: [] };
  2769→}

  2771→/**
  2772→ * CHECK RETORNO — fire when a card's power drops to ≤ its retorno keywordValue.
  2773→ * Removes the card from the board and adds it to the hand with cost 0.
  2774→ * Called from gameStore after each major board mutation (play, end_turn, start_turn, AI play).
  2775→ */
  2776→export function checkRetorno(
  2777→  board: Board,
  2778→  hand: HandCard[],
  2779→  side: "player" | "enemy",
  2780→  logs: TriggerResult["logs"] = [],
  2781→): { board: Board; hand: HandCard[] } {
  2782→  const newBoard = cloneBoard(board);
  2783→  const newHand = [...hand];
  2784→
  2785→  for (const lane of LANE_IDS) {
  2786→    const arr = newBoard[lane][side];
  2787→    for (let i = arr.length - 1; i >= 0; i--) {
  2788→      const card = arr[i];
  2789→      if (card.keyword === "retorno" && card.keywordValue && card.currentPower <= card.keywordValue) {
  2790→        // Return to hand with cost 0
  2791→        arr.splice(i, 1);
  2792→        newHand.push({
  2793→          ...card,
  2794→          uid: uuidv4(),
  2795→          cost: 0,
  2796→          handBonus: 0,
  2797→          costReduction: undefined,
  2798→          costReductionSources: undefined,
  2799→          powerLog: [],
  2800→        });
  2801→        logs.push({ side, text: `🔄 ${card.name} (Retorno): poder ${card.currentPower} ≤ ${card.keywordValue}, regresa a mano con coste 0` });
  2802→      }
  2803→    }
  2804→  }
  2805→
  2806→  return { board: newBoard, hand: newHand };
  2807→}

  2809→/**
  2810→ * Check UMBRAL triggers on the board. Fires each card whose `umbralCondition`
  2811→ * is now satisfied (and hasn't already fired). Marks the card so it doesn't
  2812→ * fire again. Returns the new board + log entries + list of fired cards.
  2813→ */
  2814→export function checkUmbralTriggers(board: Board, hand: HandCard[] = []): TriggerResult {
  2815→  const newBoard = cloneBoard(board);
  2816→  const logs: TriggerResult["logs"] = [];
  2817→  const umbralFired: TriggerResult["umbralFired"] = [];
  2818→
  2819→  for (const lane of LANE_IDS) {
  2820→    for (const side of ["player", "enemy"] as const) {
  2821→      for (const card of newBoard[lane][side]) {
  2822→        if (card.trigger !== "despertar") continue;
  2823→        if (card.umbralTriggered) continue;
  2824→        const threshold = card.umbralCondition?.powerThreshold;
  2825→        if (threshold === undefined) continue;
  2826→        if (card.currentPower >= threshold) {
  2827→          // Fire: +5 permanent bonus to self (one-shot)
  2828→          applyPowerDelta(newBoard, side, lane, card.uid, 5, card.name, "despertar");
  2829→          // applyPowerDelta REPLACES the array element with a new object (via
  2830→          // spread), so `card` now points to the OLD object. We must find the
  2831→          // new array element by uid to set umbralTriggered on it, otherwise
  2832→          // the trigger would re-fire every turn (pre-existing bug fixed here).
  2833→          const arrAfter = newBoard[lane][side];
  2834→          const idxAfter = arrAfter.findIndex((c) => c.uid === card.uid);
  2835→          if (idxAfter >= 0) {
  2836→            arrAfter[idxAfter] = { ...arrAfter[idxAfter], umbralTriggered: true };
  2837→          }
  2838→          umbralFired.push({ side, lane, uid: card.uid });
  2839→          logs.push({
  2840→            side,
  2841→            text: `🔥 ${card.name} despierta! Superó ${threshold} de poder: +5 (Despertar)`,
  2842→          });
  2843→          // Etapa 3: Frenesí fires after the umbral +5 bonus.
  2844→          // Re-read the card from the array (applyPowerDelta replaced it).
  2845→          const cardAfter = newBoard[lane][side].find((c) => c.uid === card.uid);
  2846→          if (cardAfter) {
  2847→            // ====== FIRE KEYWORD EFFECTS (despertar) ======
  2848→            // Generic keyword dispatcher — fires ANY keyword's effect when despertar triggers.
  2849→            // E.g. Tótem Sangriento (trigger=despertar, keyword=celeridad) will now
  2850→            // fire Celeridity when its power threshold is met.
  2851→            fireKeywordEffects(newBoard, side, lane, cardAfter, side === "player" ? hand : [], logs);
  2852→            triggerFrenesiIfApplicable(newBoard, side, lane, cardAfter.uid, cardAfter.name, "Despertar", logs);
  2853→          }
  2854→        }
  2855→      }
  2856→    }
  2857→  }
  2858→
  2859→  return { board: newBoard, logs, umbralFired };
  2860→}
