   207→function freshHandFromPile(pile: HandCard[], n: number): HandCard[] {
   208→  const out: HandCard[] = [];
   209→  for (let i = 0; i < n && pile.length > 0; i++) {
   210→    const card = pile.pop()!;
   211→    out.push({ ...card, uid: uuidv4(), handBonus: 0 });
   212→  }
   213→  // ensureCost1 guarantee: if the starting hand has no cost-1 card, swap in
   214→  // one from the pile (so the player can always play something on turn 1).
   215→  if (!out.some((c) => c.cost === 1) && pile.length > 0) {
   216→    const idx = pile.findIndex((c) => c.cost === 1);
   217→    if (idx >= 0) {
   218→      const replacement = pile.splice(idx, 1)[0];
   219→      // Swap out the highest-cost card to make room
   220→      let swapIdx = 0;
   221→      for (let i = 1; i < out.length; i++) {
   222→        if (out[i].cost > out[swapIdx].cost) swapIdx = i;
   223→      }
   224→      pile.push(out[swapIdx]);
   225→      out[swapIdx] = { ...replacement, uid: uuidv4(), handBonus: 0 };
   226→    }
   227→  }
   228→  return out;
   229→}

   231→function pushLog(
   232→  list: LogEntry[],
   233→  turn: number,
   234→  side: LogEntry["side"],
   235→  text: string,
   236→  kind: LogEntry["kind"] = "info"
   237→): LogEntry[] {
   238→  const entry: LogEntry = {
   239→    id: uuidv4(),
   240→    turn,
   241→    side,
   242→    text,
   243→    kind,
   244→  };
   245→  return [entry, ...list].slice(0, MAX_LOG_ENTRIES);
   246→}

   248→/**
   249→ * Push all log entries from a TriggerResult onto the game log.
   250→ * Used by the turn-lifecycle hooks (start_turn, end_turn, on_death, umbral).
   251→ */
   252→function pushTriggerLogs(
   253→  list: LogEntry[],
   254→  turn: number,
   255→  result: { logs: TriggerResult["logs"] }
   256→): LogEntry[] {
   257→  let next = list;
   258→  for (const l of result.logs) {
   259→    next = pushLog(next, turn, l.side, l.text, "info");
   260→  }
   261→  return next;
   262→}

   268→/**
   269→ * Resolve the player's deck from the loaded card pool + deck card IDs.
   270→ * Mago/Asesino siempre usan su mazo espejo; Guerrero permite custom de BD.
   271→ */
   272→function resolveDeck(state: GameStoreState): CardDef[] {
   273→  if (state.gameClass !== "guerrero") {
   274→    return CLASS_DECKS[state.gameClass];
   275→  }
   276→  const { cardPool, deckCardIds } = state;
   277→  if (cardPool.length > 0 && deckCardIds.length > 0) {
   278→    const resolved = deckCardIds
   279→      .map((id) => cardPool.find((c) => c.id === id))
   280→      .filter((c): c is CardDef => Boolean(c));
   281→    if (resolved.length === deckCardIds.length) {
   282→      return resolved;
   283→    }
   284→  }
   285→  return WARRIOR_DECK;
   286→}

   288→/**

   290→ * Used when lane bonuses are (re-)rolled — at game start (initGame) and

   295→  setTimeout(() => {
   296→    useGameStore.setState({
   297→      laneBonusFlashes: nullBonuses(ALL_LANE_IDS),
   298→    });
   299→  }, 1500);
   300→}

   330→  deckLoading: false,
   331→  deckEditorReturn: "main-menu",

   333→  draggingCardUid: null,
   334→  invalidPlayMsg: null,

   336→  log: [],

   377→      selectedCardUid: null,
   378→      invalidPlayMsg: null,
   379→      lastEnemyPlays: [],
   380→      log: pushLog(

   388→    // Trigger visual flash on each lane showing its bonus
   389→    flashLaneBonuses(laneBonuses);

   392→  selectCard: (uid) => {
   393→    const cur = get().selectedCardUid;
   394→    set({ selectedCardUid: cur === uid ? null : uid, invalidPlayMsg: null });
   395→    if (useSettings.getState().soundOn) sfx.select();
   396→  },
   397→
   398→  setDraggingCard: (uid) => {
   399→    set({ draggingCardUid: uid });
   400→  },
   401→
   402→  canPlaySelected: (lane) => {
   403→    const { selectedCardUid, hand, board, energy, laneBonuses } = get();
   404→    if (!selectedCardUid) return false;
   405→    const card = hand.find((c) => c.uid === selectedCardUid);
   406→    if (!card) return false;
   407→    const laneBonus = laneBonuses[lane];
   408→    if (!isCardCostAllowed(card.cost, laneBonus)) return false;
   409→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   410→    if (getFinalCost(baseCost, laneBonus) > energy) return false;
   411→    if (board[lane].player.length >= getLaneMaxCards(laneBonus)) return false;
   412→    return true;
   413→  },
   414→
   415→  playSelectedCardIntoLane: (lane) => {
   416→    const state = get();
   417→    if (state.phase !== "playing") return;
   418→    if (!state.selectedCardUid) return;
   419→
   420→    const card = state.hand.find((c) => c.uid === state.selectedCardUid);
   421→    if (!card) return;
   422→
   423→    const laneBonus = state.laneBonuses[lane];
   424→    const activeTrap = state.laneTraps[lane];
   425→
   426→    // Pre-flight checks with friendly messages (before playCard throws)
   427→    if (!isCardCostAllowed(card.cost, laneBonus)) {
   428→      set({
   429→        invalidPlayMsg: `"${card.name}" no se puede jugar en ${lane}: el bono exige coste exacto.`,
   430→      });
   431→      return;
   432→    }
   433→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   434→    const effCost = getFinalCost(baseCost, laneBonus);
   435→    if (effCost > state.energy) {
   436→      set({
   437→        invalidPlayMsg: `Sin energía suficiente para "${card.name}" (necesita ${effCost}${laneBonus?.kind === "cost_neg" || laneBonus?.kind === "cost_pos" ? " ajustado por bono" : ""}, tienes ${state.energy}).`,
   438→      });
   439→      return;
   440→    }
   441→    const maxCards = getLaneMaxCards(laneBonus);
   442→    if (state.board[lane].player.length >= maxCards) {
   443→      set({
   444→        invalidPlayMsg: `El carril ${lane} está lleno (máx. ${maxCards} por bando).`,
   445→      });
   446→      return;
   447→    }
   448→
   449→    try {
   450→      // Capture undo snapshot BEFORE the play
   451→      const undoSnapshot: UndoSnapshot = {
   452→        board: state.board,
   453→        hand: state.hand,
   454→        energy: state.energy,
   455→        laneTraps: state.laneTraps,
   456→        nextTurnEnergyPenalty: state.nextTurnEnergyPenalty,
   457→        gameStats: { ...state.gameStats },
   458→        playsThisTurn: state.playsThisTurn,
   459→      };
   460→
   461→      const res = playCard(
   462→        state.board,
   463→        state.hand,
   464→        state.energy,
   465→        "player",
   466→        lane,
   467→        card,
   468→        laneBonus,
   469→        activeTrap,
   470→        state.playsThisTurn + 1
   471→      );
   472→
   473→      // Sounds: card play, then buff/debuff/remove/trap as appropriate
   474→      if (useSettings.getState().soundOn) {
   475→        sfx.cardPlay();
   476→        const hasBuff = res.mutations.some((m) => m.powerDelta > 0);
   477→        const hasDebuff = res.mutations.some((m) => m.powerDelta < 0);
   478→        const hasRemove = res.removedCards.length > 0;
   479→        if (hasBuff) sfx.buff();
   480→        if (hasDebuff) sfx.debuff();
   481→        if (hasRemove) sfx.remove();
   482→        if (res.trapTriggered) sfx.trapTrigger();
   483→      }
   484→
   485→      // Floating numbers from mutations
   486→      const floatingNumbers: FloatingNumber[] = res.mutations.map((m) => ({
   487→        id: uuidv4(),
   488→        uid: m.uid,
   489→        value: m.powerDelta,
   490→      }));
   491→
   492→      // Track game stats
   493→      const stats: GameStats = { ...state.gameStats };
   494→      stats.cardsPlayedByPlayer++;
   495→      stats.buffsApplied += res.mutations.filter((m) => m.powerDelta > 0).length;
   496→      stats.debuffsToEnemy += res.mutations.filter(
   497→        (m) => m.powerDelta < 0 && m.side === "enemy"
   498→      ).length;
   499→      stats.enemyCardsDestroyed += res.removedCards.filter(
   500→        (rc) => rc.side === "enemy"
   501→      ).length;
   502→      stats.totalEnergyUsed += effCost;
   503→      // Lane turnaround: was losing or tied, now winning
   504→      const oldPlayerPower = state.board[lane].player.reduce(
   505→        (a, c) => a + c.currentPower,
   506→        0
   507→      );
   508→      const oldEnemyPower = state.board[lane].enemy.reduce(
   509→        (a, c) => a + c.currentPower,
   510→        0
   511→      );
   512→      const newPlayerPower = res.board[lane].player.reduce(
   513→        (a, c) => a + c.currentPower,
   514→        0
   515→      );
   516→      const newEnemyPower = res.board[lane].enemy.reduce(
   517→        (a, c) => a + c.currentPower,
   518→        0
   519→      );
   520→      if (oldPlayerPower <= oldEnemyPower && newPlayerPower > newEnemyPower) {
   521→        stats.laneTurnarounds++;
   522→      }
   523→
   524→      // Update lane traps: clear if triggered (consumed), set if newTrap
   525→      let newLaneTraps: LaneTraps = state.laneTraps;
   526→      if (res.trapTriggered) {
   527→        newLaneTraps = { ...newLaneTraps, [lane]: null };
   528→      }
   529→      if (res.newTrap) {
   530→        newLaneTraps = { ...newLaneTraps, [lane]: res.newTrap };
   531→      }
   532→
   533→      // Energy penalty for next turn (Sobrecarga)
   534→      let nextTurnEnergyPenalty = state.nextTurnEnergyPenalty;
   535→      if (res.energyNextTurnDelta) {
   536→        nextTurnEnergyPenalty += res.energyNextTurnDelta;
   537→      }
   538→
   539→      let log = pushLog(
   540→        state.log,
   541→        state.turn,
   542→        "player",
   543→        `Juega "${card.name}" en ${lane} (Poder ${res.playedCard.currentPower}).`,
   544→        "play"
   545→      );
   546→      for (const m of res.mutations) {
   547→        // Shield-only mutations (Amparo) log as "Escudo +X" instead of "Debuff 0"
   548→        if (m.powerDelta === 0 && m.shieldDelta !== undefined && m.shieldDelta !== 0) {
   549→          log = pushLog(
   550→            log,
   551→            state.turn,
   552→            m.side === "player" ? "player" : "enemy",
   553→            `Escudo +${m.shieldDelta} en ${m.lane}.`,
   554→            "buff"
   555→          );
   556→          continue;
   557→        }
   558→        const verb = m.powerDelta > 0 ? "buff" : "debuff";
   559→        const sign = m.powerDelta > 0 ? "+" : "";
   560→        log = pushLog(
   561→          log,
   562→          state.turn,
   563→          m.side === "player" ? "player" : "enemy",
   564→          `${verb === "buff" ? "Buff" : "Debuff"} ${sign}${m.powerDelta} en ${m.lane}.`,
   565→          verb as "buff" | "debuff"
   566→        );
   567→      }
   568→      for (const rc of res.removedCards) {
   569→        log = pushLog(
   570→          log,
   571→          state.turn,
   572→          rc.side === "player" ? "player" : "enemy",
   573→          `Carta destruida en ${rc.lane} (${rc.reason}).`,
   574→          "debuff"
   575→        );
   576→      }
   577→      if (res.trapTriggered) {
   578→        log = pushLog(
   579→          log,
   580→          state.turn,
   581→          "system",
   582→          `¡Trampa activada en ${lane}! -${activeTrap?.value ?? 0} al poder.`,
   583→          "debuff"
   584→        );
   585→      }
   586→      if (res.newTrap) {
   587→        log = pushLog(
   588→          log,
   589→          state.turn,
   590→          "player",
   591→          `Trampa colocada en ${lane}.`,
   592→          "info"
   593→        );
   594→      }
   595→
   596→      // ====== TRIGGER DISPATCH ======
   597→      // ON_DEATH: for each card destroyed by this play, fire its on_death effect.
   598→      // The dying card's full state is recovered from the pre-mutation board.
   599→      let triggerBoard = res.board;
   600→      for (const rc of res.removedCards) {
   601→        const deadCard = state.board[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   602→        if (!deadCard) continue;
   603→        const deathRes = applyOnDeathEffects(triggerBoard, rc.side, rc.lane, deadCard);
   604→        triggerBoard = deathRes.board;
   605→        log = pushTriggerLogs(log, state.turn, deathRes);
   606→      }
   607→
   608→      // ====== POWER LOG TRACKING ======
   609→      const playedUid = res.playedCard.uid;
   610→      const _playedOnBoard = triggerBoard[lane]["player" as "player" | "enemy"].find(c => c.uid === playedUid)
   611→        ?? triggerBoard[lane]["enemy" as "player" | "enemy"].find(c => c.uid === playedUid);
   612→      if (_playedOnBoard) {
   613→        const pLog: PowerLogEntry[] = [];
   614→        if (card.handBonus) pLog.push({ source: card.keyword ? KEYWORD_META[card.keyword].label : "Bonificador en mano", delta: card.handBonus, kind: "hand" });
   615→        if (card.subtypes.includes(lane)) pLog.push({ source: "Sinergia", delta: 2, kind: "synergy" });
   616→        const lbGain = getLaneBonusPower(card.category, laneBonus);
   617→        if (lbGain > 0) pLog.push({ source: laneBonus?.label ?? "Campo", delta: lbGain, kind: "field" });
   618→        const effBonus = res.playedCard.currentPower - card.power - card.handBonus - (card.subtypes.includes(lane) ? 2 : 0) - lbGain;
   619→        if (effBonus !== 0) pLog.push({ source: "Efecto propio", delta: effBonus, kind: "effect" });
   620→        if (res.trapTriggered && activeTrap) pLog.push({ source: "Trampa enemiga", delta: -activeTrap.value, kind: "trap" });
   621→        _playedOnBoard.powerLog = pLog;
   622→      }
   623→      for (const m of res.mutations) {
   624→        const arr = triggerBoard[m.lane][m.side];
   625→        const tgt = arr.find(c => c.uid === m.uid);
   626→        if (tgt) {
   627→          if (!tgt.powerLog) tgt.powerLog = [];
   628→          tgt.powerLog.push({ source: card.name, delta: m.powerDelta, kind: m.powerDelta > 0 ? "buff" : "debuff" });
   629→        }
   630→      }
   631→
   632→      set({
   633→        board: triggerBoard,
   634→        hand: res.hand,
   635→        energy: res.remainingEnergy,
   636→        laneTraps: newLaneTraps,
   637→        nextTurnEnergyPenalty,
   638→        floatingNumbers,
   639→        gameStats: stats,
   640→        undoSnapshot,
   641→        playsThisTurn: state.playsThisTurn + 1,
   642→        legendaryBurst: card.rarity === "legendary",
   643→        selectedCardUid: null,
   644→        invalidPlayMsg: null,
   645→        log,
   646→      });
   647→
   648→      // Clear floating numbers after they've had time to animate
   649→      setTimeout(() => {
   650→        get().clearFloatingNumbers();
   651→      }, 1500);
   652→
   653→      // Clear legendary burst after the animation completes
   654→      if (card.rarity === "legendary") {
   655→        setTimeout(() => {
   656→          set({ legendaryBurst: false });
   657→        }, 1900);
   658→      }
   659→    } catch (e) {
   660→      set({
   661→        invalidPlayMsg: (e as Error).message,
   662→      });
   663→    }
   664→  },
   665→
   666→  endTurn: async () => {
   667→    const state = get();
   668→    if (state.phase !== "playing") return;
   669→
   670→    // Pasar button click sound
   671→    if (useSettings.getState().soundOn) sfx.click();
   672→
   673→    // 0. Show banner, freeze the board
   674→    set({
   675→      aiTurnPhase: "banner",
   676→      phase: "enemy",
   677→      selectedCardUid: null,
   678→      invalidPlayMsg: null,
   679→    });
   680→    if (useSettings.getState().soundOn) sfx.turnPass();
   681→    await sleep(1500);
   682→
   683→    // 1. Switch to "thinking" phase
   684→    set({ aiTurnPhase: "thinking" });
   685→    if (useSettings.getState().soundOn) sfx.enemyTurn();
   686→    await sleep(450);
   687→
   688→    // Re-read state after the async waits
   689→    const freshState = get();
   690→
   691→    // ====== TRIGGER: END_TURN (player) ======
   692→    // Player just pressed "Pasar" — fire their end_turn effects before AI plays.
   693→    const playerEndRes = applyEndTurnEffects(freshState.board, "player", freshState.hand, freshState.activeLanes);
   694→    let triggerBoard = playerEndRes.board;
   695→    let triggerLogs: LogEntry[] = pushTriggerLogs(freshState.log, freshState.turn, playerEndRes);
   696→
   697→    // ====== TRIGGER: START_TURN (enemy) ======
   698→    const enemyStartRes = applyStartTurnEffects(triggerBoard, "enemy", [], freshState.activeLanes);
   699→    triggerBoard = enemyStartRes.board;
   700→    triggerLogs = pushTriggerLogs(triggerLogs, freshState.turn, enemyStartRes);
   701→
   702→    // 2. Build the enemy hand: draw 3 random cards from a fresh shuffled pile.
   703→    const resolvedDeck = resolveDeck(freshState);
   704→    const enemyEnergy = freshState.maxEnergy;
   705→    const enemyHand: HandCard[] = [];
   706→    const deckPool = [...buildDrawPileFrom(resolvedDeck)];
   707→    for (let i = 0; i < 3 && deckPool.length > 0; i++) {
   708→      const idx = Math.floor(Math.random() * deckPool.length);
   709→      const c = deckPool.splice(idx, 1)[0];
   710→      enemyHand.push({ ...c, uid: uuidv4(), handBonus: 0 });
   711→    }
   712→
   713→    // 3. Plan and execute the AI plays on the actual board, respecting lane
   714→    //    bonuses AND lane traps. Use the post-trigger board as starting point.
   715→    const plan = planEnemyTurn(
   716→      triggerBoard,
   717→      enemyHand,
   718→      enemyEnergy,
   719→      freshState.laneBonuses,
   720→      freshState.laneTraps,
   721→      freshState.activeLanes
   722→    );
   723→    let workBoard = triggerBoard;
   724→    let workEnemyHand = [...enemyHand];
   725→    let workEnergy = enemyEnergy;
   726→    let workTraps: LaneTraps = { ...freshState.laneTraps };
   727→    let log = triggerLogs;
   728→    let gameStats: GameStats = { ...freshState.gameStats };
   729→
   730→    for (const play of plan) {
   731→      try {
   732→        const prePlayBoard = workBoard;
   733→        const res = playCard(
   734→          workBoard,
   735→          workEnemyHand,
   736→          workEnergy,
   737→          "enemy",
   738→          play.lane,
   739→          play.card,
   740→          freshState.laneBonuses[play.lane],
   741→          workTraps[play.lane]
   742→        );
   743→        workBoard = res.board;
   744→        workEnemyHand = res.hand;
   745→        workEnergy = res.remainingEnergy;
   746→
   747→        // Update traps: clear if triggered, set if newTrap
   748→        if (res.trapTriggered) {
   749→          workTraps = { ...workTraps, [play.lane]: null };
   750→        }
   751→        if (res.newTrap) {
   752→          workTraps = { ...workTraps, [play.lane]: res.newTrap };
   753→        }
   754→
   755→        // Sound: enemy card play + buffs/debuffs/remove/trap
   756→        if (useSettings.getState().soundOn) {
   757→          sfx.cardPlay();
   758→          if (res.mutations.some((m) => m.powerDelta > 0)) sfx.buff();
   759→          if (res.mutations.some((m) => m.powerDelta < 0)) sfx.debuff();
   760→          if (res.removedCards.length > 0) sfx.remove();
   761→          if (res.trapTriggered) sfx.trapTrigger();
   762→        }
   763→
   764→        // Enemy stats
   765→        gameStats.cardsPlayedByEnemy++;
   766→        gameStats.buffsApplied += res.mutations.filter(
   767→          (m) => m.powerDelta > 0
   768→        ).length;
   769→        // debuffs to PLAYER cards count as "debuffs to enemy" from enemy's
   770→        // perspective — we still keep a single "debuffsToEnemy" counter that
   771→        // tracks debuffs applied TO the enemy side, so for enemy plays we
   772→        // don't increment it (the enemy isn't debuffing itself).
   773→        // (Player cards destroyed by enemy plays don't increment
   774→        // `enemyCardsDestroyed` either — that stat is player-centric.)
   775→
   776→        log = pushLog(
   777→          log,
   778→          freshState.turn,
   779→          "enemy",
   780→          `Rival juega "${play.card.name}" en ${play.lane} (Poder ${res.playedCard.currentPower}).`,
   781→          "play"
   782→        );
   783→        for (const m of res.mutations) {
   784→          // Shield-only mutations (Amparo) log as "Escudo +X" instead of "0"
   785→          if (m.powerDelta === 0 && m.shieldDelta !== undefined && m.shieldDelta !== 0) {
   786→            log = pushLog(
   787→              log,
   788→              freshState.turn,
   789→              m.side === "player" ? "player" : "enemy",
   790→              `Escudo +${m.shieldDelta} en ${m.lane} (${m.side === "player" ? "tú" : "rival"}).`,
   791→              "buff"
   792→            );
   793→            continue;
   794→          }
   795→          const sign = m.powerDelta > 0 ? "+" : "";
   796→          log = pushLog(
   797→            log,
   798→            freshState.turn,
   799→            m.side === "player" ? "player" : "enemy",
   800→            `${sign}${m.powerDelta} en ${m.lane} (${m.side === "player" ? "tú" : "rival"}).`,
   801→            m.powerDelta > 0 ? "buff" : "debuff"
   802→          );
   803→        }
   804→        for (const rc of res.removedCards) {
   805→          log = pushLog(
   806→            log,
   807→            freshState.turn,
   808→            rc.side === "player" ? "player" : "enemy",
   809→            `Carta destruida en ${rc.lane} (${rc.reason}).`,
   810→            "debuff"
   811→          );
   812→        }
   813→        if (res.trapTriggered) {
   814→          log = pushLog(
   815→            log,
   816→            freshState.turn,
   817→            "system",
   818→            `¡Trampa enemiga activada en ${play.lane}!`,
   819→            "debuff"
   820→          );
   821→        }
   822→        if (res.newTrap) {
   823→          log = pushLog(
   824→            log,
   825→            freshState.turn,
   826→            "enemy",
   827→            `Trampa enemiga colocada en ${play.lane}.`,
   828→            "info"
   829→          );
   830→        }
   831→
   832→        // ====== TRIGGER DISPATCH (enemy play) ======
   833→        // ON_DEATH: for each card destroyed by the enemy's play.
   834→        // Look up the dying card in the PRE-play board (prePlayBoard) because
   835→        // workBoard (= res.board) has already had removedCards filtered out.
   836→        for (const rc of res.removedCards) {
   837→          const deadCard = prePlayBoard[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   838→          if (!deadCard) continue;
   839→          const deathRes = applyOnDeathEffects(workBoard, rc.side, rc.lane, deadCard);
   840→          workBoard = deathRes.board;
   841→          log = pushTriggerLogs(log, freshState.turn, deathRes);
   842→        }
   843→      } catch {
   844→        // ignore failed plays
   845→      }
   846→    }
   847→
   848→    // ====== TRIGGER: END_TURN (enemy) ======
   849→    // Enemy just finished playing — fire their end_turn effects before turn advances.
   850→    const enemyEndRes = applyEndTurnEffects(workBoard, "enemy", [], freshState.activeLanes);
   851→    workBoard = enemyEndRes.board;
   852→    log = pushTriggerLogs(log, freshState.turn, enemyEndRes);
   853→
   854→    // 4. Advance turn, apply Sobrecarga energy penalty
   855→    const newTurn = freshState.turn + 1;
   856→    const newMaxEnergy = Math.min(MAX_TURNS, freshState.maxEnergy + 1);
   857→    const newEnergy = Math.max(0, newMaxEnergy + freshState.nextTurnEnergyPenalty);
   858→
   859→    if (newTurn > MAX_TURNS) {
   860→      // Game over
   861→      const finalResult = computeFinalResult(workBoard, freshState.activeLanes);
   862→      log = pushLog(
   863→        log,
   864→        MAX_TURNS,
   865→        "system",
   866→        `Fin del juego: ${finalResult.result === "WIN" ? "¡Victoria!" : finalResult.result === "LOSS" ? "Derrota" : "Empate"} (${finalResult.playerWins}-${finalResult.enemyWins}, E ${finalResult.draws}).`,
   867→        "end"
   868→      );
   869→      if (useSettings.getState().soundOn) {
   870→        if (finalResult.result === "WIN") sfx.victory();
   871→        else if (finalResult.result === "LOSS") sfx.defeat();
   872→      }
   873→      set({
   874→        board: workBoard,
   875→        phase: "ended",
   876→        result: finalResult.result,
   877→        finalResult,
   878→        lastEnemyPlays: plan,
   879→        laneTraps: workTraps,
   880→        nextTurnEnergyPenalty: 0,
   881→        aiTurnPhase: "idle",
   882→        log,
   883→        gameStats,
   884→        playsThisTurn: 0,
   885→      });
   886→      return;
   887→    }
   888→
   889→    // 5. Draw a card, refill energy, etc.
   890→    const newDrawPile = [...freshState.drawPile];
   891→    let newHand = [...freshState.hand];
   892→    if (newHand.length < MAX_HAND_SIZE && newDrawPile.length > 0) {
   893→      const drawn = newDrawPile.pop()!;
   894→      newHand.push({ ...drawn, uid: uuidv4(), handBonus: 0 });
   895→    }
   896→
   897→    log = pushLog(
   898→      log,
   899→      newTurn,
   900→      "system",
   901→      `Turno ${newTurn} — Energía ${newEnergy}/${newMaxEnergy}${freshState.nextTurnEnergyPenalty ? ` (penalización ${freshState.nextTurnEnergyPenalty})` : ""}.`,
   902→      "info"
   903→    );
   904→
   905→    // ====== TRIGGER: START_TURN (player) ======
   906→    // Player's new turn begins — fire their start_turn effects before they play.
   907→    const playerStartRes = applyStartTurnEffects(workBoard, "player", newHand, freshState.activeLanes);
   908→    workBoard = playerStartRes.board;
   909→    log = pushTriggerLogs(log, newTurn, playerStartRes);
   910→
   911→    // ====== RE-ROLL LANE BONUSES AT TURN 4 ======
   912→    // User spec: at the start of turn 4, do another random roll of lane bonuses
   913→    // so the player needs to adapt their strategy mid-match (was static before).
   914→    let newLaneBonuses = freshState.laneBonuses;
   915→    if (newTurn === 4) {
   916→      newLaneBonuses = rollLaneBonuses(freshState.activeLanes);
   917→      const bonusSummary = freshState.activeLanes.map((ln) => {
   918→        const b = newLaneBonuses[ln];
   919→        return b ? `${ln}: ${b.label}` : `${ln}: sin bono`;
   920→      }).join(" · ");
   921→      log = pushLog(log, newTurn, "system", `🔮 ¡Bonos re-rollados! ${bonusSummary}`, "buff");
   922→      // Trigger visual flash on each lane showing the new bonus
   923→      flashLaneBonuses(newLaneBonuses);
   924→    }
   925→
   926→    set({
   927→      board: workBoard,
   928→      turn: newTurn,
   929→      energy: newEnergy,
   930→      maxEnergy: newMaxEnergy,
   931→      drawPile: newDrawPile,
   932→      hand: newHand,
   933→      phase: "playing",
   934→      selectedCardUid: null,
   935→      invalidPlayMsg: null,
   936→      lastEnemyPlays: plan,
   937→      laneTraps: workTraps,
   938→      laneBonuses: newLaneBonuses,
   939→      nextTurnEnergyPenalty: 0,
   940→      aiTurnPhase: "idle",
   941→      log,
   942→      gameStats,
   943→      playsThisTurn: 0,
   944→    });
   945→  },

  1015→  swapCardInDeck: (index, cardId) => {
  1016→    // Allow placing a card into ANY of the 12 deck slots, even if the slot is
  1017→    // beyond the current deckCardIds length (e.g. user emptied the deck then
  1018→    // clicks slot 5 — we pad with empty strings up to that index).
  1019→    if (index < 0 || index >= 12) return;
  1020→    const { deckCardIds } = get();
  1021→    if (deckCardIds[index] === cardId) return;
  1022→    // Prevent duplicates (deck is 12 unique cards) — if the card is already in
  1023→    // the deck at a DIFFERENT slot, move it from the old slot to the new one
  1024→    // (old slot becomes empty). This is the "drag-to-rearrange" UX.
  1025→    const next = [...deckCardIds];
  1026→    // Pad with empty strings up to the target index so we can place at any slot
  1027→    while (next.length <= index) next.push("");
  1028→    if (deckCardIds.includes(cardId)) {
  1029→      const existingIdx = deckCardIds.indexOf(cardId);
  1030→      next[existingIdx] = "";  // clear old slot
  1031→    }
  1032→    next[index] = cardId;
  1033→    // Always store exactly 12 entries (pad/truncate to 12)
  1034→    while (next.length < 12) next.push("");
  1035→    set({ deckCardIds: next.slice(0, 12), deckDirty: true, deckSaveStatus: "idle" });
  1036→  },

  1038→  saveDeck: async () => {
  1039→    const { deckCardIds, gameClass } = get();
  1040→    if (deckCardIds.length !== 12) return;
  1041→    set({ deckSaveStatus: "saving" });
  1042→    try {
  1043→      const storageKey = gameClass === "guerrero" ? "guerrero-tactico-deck" : `guerrero-tactico-deck-${gameClass}`;
  1044→      // Save to localStorage (no API route needed — works in static export / APK)
  1045→      localStorage.setItem(storageKey, JSON.stringify({ cardIds: deckCardIds }));
  1046→      set({ deckSaveStatus: "ok", deckDirty: false });
  1047→      setTimeout(() => {
  1048→        if (get().deckSaveStatus === "ok") set({ deckSaveStatus: "idle" });
  1049→      }, 2000);
  1050→    } catch (e) {
  1051→      console.error("saveDeck error:", e);
  1052→      set({ deckSaveStatus: "err" });
  1053→    }
  1054→  },
  1055→
  1056→  resetDeck: async () => {
  1057→    set({ deckSaveStatus: "saving" });
  1058→    try {

  1062→      await get().loadDeckAndPool();
  1063→      set({ deckSaveStatus: "idle", deckDirty: false });
  1064→    } catch (e) {
  1065→      console.error("resetDeck error:", e);
  1066→      set({ deckSaveStatus: "err" });
  1067→    }
  1068→  },

  1071→// Convenience selector for lane scores
  1072→export function laneScores(board: Board, lane: LaneId) {
  1073→  const p = board[lane].player.reduce((a, c) => a + c.currentPower, 0);
  1074→  const e = board[lane].enemy.reduce((a, c) => a + c.currentPower, 0);
  1075→  return { player: p, enemy: e };
  1076→}
