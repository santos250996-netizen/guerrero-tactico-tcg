   200→   200→function freshHandFromPile(pile: HandCard[], n: number): HandCard[] {
   201→   201→  const out: HandCard[] = [];
   202→   202→  for (let i = 0; i < n && pile.length > 0; i++) {
   203→   203→    const card = pile.pop()!;
   204→   204→    out.push({ ...card, uid: uuidv4(), handBonus: 0 });
   205→   205→  }
   206→   206→  // ensureCost1 guarantee: if the starting hand has no cost-1 card, swap in
   207→   207→  // one from the pile (so the player can always play something on turn 1).
   208→   208→  if (!out.some((c) => c.cost === 1) && pile.length > 0) {
   209→   209→    const idx = pile.findIndex((c) => c.cost === 1);
   210→   210→    if (idx >= 0) {
   211→   211→      const replacement = pile.splice(idx, 1)[0];
   212→   212→      // Swap out the highest-cost card to make room
   213→   213→      let swapIdx = 0;
   214→   214→      for (let i = 1; i < out.length; i++) {
   215→   215→        if (out[i].cost > out[swapIdx].cost) swapIdx = i;
   216→   216→      }
   217→   217→      pile.push(out[swapIdx]);
   218→   218→      out[swapIdx] = { ...replacement, uid: uuidv4(), handBonus: 0 };
   219→   219→    }
   220→   220→  }
   221→   221→  return out;
   222→   222→}

   261→   261→/**
   262→   262→ * Resolve the player's deck from the loaded card pool + deck card IDs.
   263→   263→ * Falls back to WARRIOR_DECK if the pool/deck isn't loaded or has missing cards.
   264→   264→ */
   265→   265→function resolveDeck(state: GameStoreState): CardDef[] {
   266→   266→  const { cardPool, deckCardIds } = state;
   267→   267→  if (cardPool.length > 0 && deckCardIds.length > 0) {
   268→   268→    const resolved = deckCardIds
   269→   269→      .map((id) => cardPool.find((c) => c.id === id))
   270→   270→      .filter((c): c is CardDef => Boolean(c));
   271→   271→    if (resolved.length === deckCardIds.length) {
   272→   272→      return resolved;
   273→   273→    }
   274→   274→  }
   275→   275→  return WARRIOR_DECK;
   276→   276→}

   397→   397→  playSelectedCardIntoLane: (lane) => {
   398→   398→    const state = get();
   399→   399→    if (state.phase !== "playing") return;
   400→   400→    if (!state.selectedCardUid) return;
   401→   401→
   402→   402→    const card = state.hand.find((c) => c.uid === state.selectedCardUid);
   403→   403→    if (!card) return;
   404→   404→
   405→   405→    const laneBonus = state.laneBonuses[lane];
   406→   406→    const activeTrap = state.laneTraps[lane];
   407→   407→
   408→   408→    // Pre-flight checks with friendly messages (before playCard throws)
   409→   409→    if (!isCardCostAllowed(card.cost, laneBonus)) {
   410→   410→      set({
   411→   411→        invalidPlayMsg: `"${card.name}" no se puede jugar en ${lane}: el bono exige coste exacto.`,
   412→   412→      });
   413→   413→      return;
   414→   414→    }
   415→   415→    // Etapa 4: account for Celeridad costReduction on the hand card.
   416→   416→    const baseCost = Math.max(1, card.cost - (card.costReduction ?? 0));
   417→   417→    const effCost = getFinalCost(baseCost, laneBonus, card.effectId, card.keywordValue);
   418→   418→    if (effCost > state.energy) {
   419→   419→      set({
   420→   420→        invalidPlayMsg: `Sin energía suficiente para "${card.name}" (necesita ${effCost}${laneBonus?.kind === "cost_neg" || laneBonus?.kind === "cost_pos" ? " ajustado por bono" : ""}${card.effectId === "sobrecarga" ? ` (Sobrecarga: -${card.keywordValue ?? 1})` : ""}, tienes ${state.energy}).`,
   421→   421→      });
   422→   422→      return;
   423→   423→    }
   424→   424→    const maxCards = getLaneMaxCards(laneBonus);
   425→   425→    if (state.board[lane].player.length >= maxCards) {
   426→   426→      set({
   427→   427→        invalidPlayMsg: `El carril ${lane} está lleno (máx. ${maxCards} por bando).`,
   428→   428→      });
   429→   429→      return;
   430→   430→    }
   431→   431→
   432→   432→    try {
   433→   433→      // Capture undo snapshot BEFORE the play
   434→   434→      const undoSnapshot: UndoSnapshot = {
   435→   435→        board: state.board,
   436→   436→        hand: state.hand,
   437→   437→        energy: state.energy,
   438→   438→        laneTraps: state.laneTraps,
   439→   439→        nextTurnEnergyPenalty: state.nextTurnEnergyPenalty,
   440→   440→        gameStats: { ...state.gameStats },
   441→   441→      };
   442→   442→
   443→   443→      const res = playCard(
   444→   444→        state.board,
   445→   445→        state.hand,
   446→   446→        state.energy,
   447→   447→        "player",
   448→   448→        lane,
   449→   449→        card,
   450→   450→        laneBonus,
   451→   451→        activeTrap
   452→   452→      );
   453→   453→
   454→   454→      // Sounds: card play, then buff/debuff/remove/trap as appropriate
   455→   455→      if (useSettings.getState().soundOn) {
   456→   456→        sfx.cardPlay();
   457→   457→        const hasBuff = res.mutations.some((m) => m.powerDelta > 0);
   458→   458→        const hasDebuff = res.mutations.some((m) => m.powerDelta < 0);
   459→   459→        const hasRemove = res.removedCards.length > 0;
   460→   460→        if (hasBuff) sfx.buff();
   461→   461→        if (hasDebuff) sfx.debuff();
   462→   462→        if (hasRemove) sfx.remove();
   463→   463→        if (res.trapTriggered) sfx.trapTrigger();
   464→   464→      }
   465→   465→
   466→   466→      // Floating numbers from mutations
   467→   467→      const floatingNumbers: FloatingNumber[] = res.mutations.map((m) => ({
   468→   468→        id: uuidv4(),
   469→   469→        uid: m.uid,
   470→   470→        value: m.powerDelta,
   471→   471→      }));
   472→   472→
   473→   473→      // Track game stats
   474→   474→      const stats: GameStats = { ...state.gameStats };
   475→   475→      stats.cardsPlayedByPlayer++;
   476→   476→      stats.buffsApplied += res.mutations.filter((m) => m.powerDelta > 0).length;
   477→   477→      stats.debuffsToEnemy += res.mutations.filter(
   478→   478→        (m) => m.powerDelta < 0 && m.side === "enemy"
   479→   479→      ).length;
   480→   480→      stats.enemyCardsDestroyed += res.removedCards.filter(
   481→   481→        (rc) => rc.side === "enemy"
   482→   482→      ).length;
   483→   483→      stats.totalEnergyUsed += effCost;
   484→   484→      // Lane turnaround: was losing or tied, now winning
   485→   485→      const oldPlayerPower = state.board[lane].player.reduce(
   486→   486→        (a, c) => a + c.currentPower,
   487→   487→        0
   488→   488→      );
   489→   489→      const oldEnemyPower = state.board[lane].enemy.reduce(
   490→   490→        (a, c) => a + c.currentPower,
   491→   491→        0
   492→   492→      );
   493→   493→      const newPlayerPower = res.board[lane].player.reduce(
   494→   494→        (a, c) => a + c.currentPower,
   495→   495→        0
   496→   496→      );
   497→   497→      const newEnemyPower = res.board[lane].enemy.reduce(
   498→   498→        (a, c) => a + c.currentPower,
   499→   499→        0
   500→   500→      );
   501→   501→      if (oldPlayerPower <= oldEnemyPower && newPlayerPower > newEnemyPower) {
   502→   502→        stats.laneTurnarounds++;
   503→   503→      }
   504→   504→
   505→   505→      // Update lane traps: clear if triggered (consumed), set if newTrap
   506→   506→      let newLaneTraps: LaneTraps = state.laneTraps;
   507→   507→      if (res.trapTriggered) {
   508→   508→        newLaneTraps = { ...newLaneTraps, [lane]: null };
   509→   509→      }
   510→   510→      if (res.newTrap) {
   511→   511→        newLaneTraps = { ...newLaneTraps, [lane]: res.newTrap };
   512→   512→      }
   513→   513→
   514→   514→      // Energy penalty for next turn (Sobrecarga)
   515→   515→      let nextTurnEnergyPenalty = state.nextTurnEnergyPenalty;
   516→   516→      if (res.energyNextTurnDelta) {
   517→   517→        nextTurnEnergyPenalty += res.energyNextTurnDelta;
   518→   518→      }
   519→   519→
   520→   520→      let log = pushLog(
   521→   521→        state.log,
   522→   522→        state.turn,
   523→   523→        "player",
   524→   524→        `Juega "${card.name}" en ${lane} (Poder ${res.playedCard.currentPower}).`,
   525→   525→        "play"
   526→   526→      );
   527→   527→      for (const m of res.mutations) {
   528→   528→        const verb = m.powerDelta > 0 ? "buff" : "debuff";
   529→   529→        const sign = m.powerDelta > 0 ? "+" : "";
   530→   530→        log = pushLog(
   531→   531→          log,
   532→   532→          state.turn,
   533→   533→          m.side === "player" ? "player" : "enemy",
   534→   534→          `${verb === "buff" ? "Buff" : "Debuff"} ${sign}${m.powerDelta} en ${m.lane}.`,
   535→   535→          verb as "buff" | "debuff"
   536→   536→        );
   537→   537→      }
   538→   538→      for (const rc of res.removedCards) {
   539→   539→        log = pushLog(
   540→   540→          log,
   541→   541→          state.turn,
   542→   542→          rc.side === "player" ? "player" : "enemy",
   543→   543→          `Carta destruida en ${rc.lane} (${rc.reason}).`,
   544→   544→          "debuff"
   545→   545→        );
   546→   546→      }
   547→   547→      for (const mc of res.movedCards) {
   548→   548→        log = pushLog(
   549→   549→          log,
   550→   550→          state.turn,
   551→   551→          "player",
   552→   552→          `Aliado movido de ${mc.fromLane} a ${mc.toLane}.`,
   553→   553→          "info"
   554→   554→        );
   555→   555→      }
   556→   556→      if (res.trapTriggered) {
   557→   557→        log = pushLog(
   558→   558→          log,
   559→   559→          state.turn,
   560→   560→          "system",
   561→   561→          `¡Trampa activada en ${lane}! -${activeTrap?.value ?? 0} al poder.`,
   562→   562→          "debuff"
   563→   563→        );
   564→   564→      }
   565→   565→      if (res.newTrap) {
   566→   566→        log = pushLog(
   567→   567→          log,
   568→   568→          state.turn,
   569→   569→          "player",
   570→   570→          `Trampa colocada en ${lane}.`,
   571→   571→          "info"
   572→   572→        );
   573→   573→      }
   574→   574→
   575→   575→      // ====== TRIGGER DISPATCH ======
   576→   576→      // 1. ON_DEATH: for each card destroyed by this play, fire its on_death effect.
   577→   577→      //    We need the dying card's full state (including trigger/effectId) which
   578→   578→      //    we can recover from the pre-mutation board (state.board).
   579→   579→      let triggerBoard = res.board;
   580→   580→      for (const rc of res.removedCards) {
   581→   581→        const deadCard = state.board[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   582→   582→        if (!deadCard) continue;
   583→   583→        const deathRes = applyOnDeathEffects(triggerBoard, rc.side, rc.lane, deadCard);
   584→   584→        triggerBoard = deathRes.board;
   585→   585→        log = pushTriggerLogs(log, state.turn, deathRes);
   586→   586→
   587→   587→        // ====== ETAPA 4: emit card_destroyed event ======
   588→   588→        // Reaction cards on the same side as the destroyed card may respond
   589→   589→        // (e.g. an ally of the sacrificed card could retaliate — future keyword).
   590→   590→        const destroyEvent: CardDestroyedEvent = {
   591→   591→          type: "card_destroyed",
   592→   592→          side: rc.side,
   593→   593→          lane: rc.lane,
   594→   594→          cardUid: rc.uid,
   595→   595→          cardName: deadCard.name,
   596→   596→          cardCategory: deadCard.category,
   597→   597→          reason: rc.reason,
   598→   598→        };
   599→   599→        const rxnRes = applyReactionEffects(triggerBoard, rc.side, destroyEvent);
   600→   600→        log = pushTriggerLogs(log, state.turn, rxnRes);
   601→   601→      }
   602→   602→      // 2. UMBRAL: check all umbral triggers on the resulting board.
   603→   603→      const umbralRes = checkUmbralTriggers(triggerBoard, res.hand);
   604→   604→      triggerBoard = umbralRes.board;
   605→   605→      log = pushTriggerLogs(log, state.turn, umbralRes);
   606→   606→
   607→   607→      // ====== CHECK RETORNO (power-drop trigger) ======
   608→   608→      // After all effects from the play, check if any card's power dropped to ≤ its
   609→   609→      // retorno keywordValue. If so, it returns to hand with cost 0.
   610→   610→      const retornoLogs: { side: "player" | "enemy" | "system"; text: string }[] = [];
   611→   611→      const retornoRes = checkRetorno(triggerBoard, res.hand, "player", retornoLogs);
   612→   612→      triggerBoard = retornoRes.board;
   613→   613→      res.hand = retornoRes.hand;
   614→   614→      log = pushTriggerLogs(log, state.turn, { logs: retornoLogs });
   615→   615→
   616→   616→      // ====== POWER LOG TRACKING ======
   617→   617→      const playedUid = res.playedCard.uid;
   618→   618→      const _playedOnBoard = triggerBoard[lane]["player" as "player" | "enemy"].find(c => c.uid === playedUid)
   619→   619→        ?? triggerBoard[lane]["enemy" as "player" | "enemy"].find(c => c.uid === playedUid);
   620→   620→      if (_playedOnBoard) {
   621→   621→        const pLog: PowerLogEntry[] = [];
   622→   622→        if (card.handBonus) pLog.push({ source: card.keyword ? KEYWORD_META[card.keyword].label : "Bonificador en mano", delta: card.handBonus, kind: "hand" });
   623→   623→        if (card.subtypes.includes(lane)) pLog.push({ source: "Sinergia", delta: 2, kind: "synergy" });
   624→   624→        const lbGain = getLaneBonusPower(card.category, laneBonus);
   625→   625→        if (lbGain > 0) pLog.push({ source: laneBonus?.label ?? "Campo", delta: lbGain, kind: "field" });
   626→   626→        const effBonus = res.playedCard.currentPower - card.power - card.handBonus - (card.subtypes.includes(lane) ? 2 : 0) - lbGain;
   627→   627→        if (effBonus !== 0) pLog.push({ source: "Efecto propio", delta: effBonus, kind: "effect" });
   628→   628→        if (res.trapTriggered && activeTrap) pLog.push({ source: "Trampa enemiga", delta: -activeTrap.value, kind: "trap" });
   629→   629→        _playedOnBoard.powerLog = pLog;
   630→   630→      }
   631→   631→      for (const m of res.mutations) {
   632→   632→        const arr = triggerBoard[m.lane][m.side];
   633→   633→        const tgt = arr.find(c => c.uid === m.uid);
   634→   634→        if (tgt) {
   635→   635→          if (!tgt.powerLog) tgt.powerLog = [];
   636→   636→          tgt.powerLog.push({ source: card.name, delta: m.powerDelta, kind: m.powerDelta > 0 ? "buff" : "debuff" });
   637→   637→        }
   638→   638→      }
   639→   639→
   640→   640→      // ====== ETAPA 4: emit card_played event ======
   641→   641→      // Reaction cards on the ENEMY side may respond to the player's card_played
   642→   642→      // (e.g. an enemy Centinela could gain +X when the player plays an Ataque).
   643→   643→      // applyReactionEffects mutates triggerBoard in place — no return value.
   644→   644→      const playedEvent: CardPlayedEvent = {
   645→   645→        type: "card_played",
   646→   646→        side: "player",
   647→   647→        lane,
   648→   648→        cardUid: res.playedCard.uid,
   649→   649→        cardName: card.name,
   650→   650→        cardCategory: card.category,
   651→   651→        cardSubtypes: card.subtypes,
   652→   652→      };
   653→   653→      const rxnRes = applyReactionEffects(triggerBoard, "enemy", playedEvent);
   654→   654→      log = pushTriggerLogs(log, state.turn, rxnRes);
   655→   655→
   656→   656→      // ====== Push keyword effect logs (celeridad target, eco target, etc.) ======
   657→   657→      if (res.kwLogs && res.kwLogs.length > 0) {
   658→   658→        log = pushTriggerLogs(log, state.turn, { logs: res.kwLogs });
   659→   659→      }
   660→   660→
   661→   661→      set({
   662→   662→        board: triggerBoard,
   663→   663→        hand: res.hand,
   664→   664→        energy: res.remainingEnergy,
   665→   665→        laneTraps: newLaneTraps,
   666→   666→        nextTurnEnergyPenalty,
   667→   667→        floatingNumbers,
   668→   668→        gameStats: stats,
   669→   669→        undoSnapshot,
   670→   670→        legendaryBurst: card.rarity === "legendary",
   671→   671→        selectedCardUid: null,
   672→   672→        invalidPlayMsg: null,
   673→   673→        log,
   674→   674→      });
   675→   675→
   676→   676→      // Clear floating numbers after they've had time to animate
   677→   677→      setTimeout(() => {
   678→   678→        get().clearFloatingNumbers();
   679→   679→      }, 1500);
   680→   680→
   681→   681→      // Clear legendary burst after the animation completes
   682→   682→      if (card.rarity === "legendary") {
   683→   683→        setTimeout(() => {
   684→   684→          set({ legendaryBurst: false });
   685→   685→        }, 1900);
   686→   686→      }
   687→   687→    } catch (e) {
   688→   688→      set({
   689→   689→        invalidPlayMsg: (e as Error).message,
   690→   690→      });
   691→   691→    }
   692→   692→  },
   693→   693→
   694→   694→  endTurn: async () => {
   695→   695→    const state = get();
   696→   696→    if (state.phase !== "playing") return;
   697→   697→
   698→   698→    // Pasar button click sound
   699→   699→    if (useSettings.getState().soundOn) sfx.click();
   700→   700→
   701→   701→    // 0. Show banner, freeze the board
   702→   702→    set({
   703→   703→      aiTurnPhase: "banner",
   704→   704→      phase: "enemy",
   705→   705→      selectedCardUid: null,
   706→   706→      invalidPlayMsg: null,
   707→   707→    });
   708→   708→    if (useSettings.getState().soundOn) sfx.turnPass();
   709→   709→    await sleep(1500);
   710→   710→
   711→   711→    // 1. Switch to "thinking" phase
   712→   712→    set({ aiTurnPhase: "thinking" });
   713→   713→    if (useSettings.getState().soundOn) sfx.enemyTurn();
   714→   714→    await sleep(450);
   715→   715→
   716→   716→    // Re-read state after the async waits
   717→   717→    const freshState = get();
   718→   718→
   719→   719→    // ====== TRIGGER: END_TURN (player) ======
   720→   720→    // Player just pressed "Pasar" — fire their end_turn effects before AI plays.
   721→   721→    const playerEndRes = applyEndTurnEffects(freshState.board, "player", freshState.hand);
   722→   722→    let triggerBoard = playerEndRes.board;
   723→   723→    let triggerLogs: LogEntry[] = pushTriggerLogs(freshState.log, freshState.turn, playerEndRes);
   724→   724→    // UMBRAL check after player's end_turn effects
   725→   725→    const playerEndUmbral = checkUmbralTriggers(triggerBoard, freshState.hand);
   726→   726→    triggerBoard = playerEndUmbral.board;
   727→   727→    triggerLogs = pushTriggerLogs(triggerLogs, freshState.turn, playerEndUmbral);
   728→   728→
   729→   729→    // ====== TRIGGER: START_TURN (enemy) ======
   730→   730→    const enemyStartRes = applyStartTurnEffects(triggerBoard, "enemy");
   731→   731→    triggerBoard = enemyStartRes.board;
   732→   732→    triggerLogs = pushTriggerLogs(triggerLogs, freshState.turn, enemyStartRes);
   733→   733→    const enemyStartUmbral = checkUmbralTriggers(triggerBoard, freshState.hand);
   734→   734→    triggerBoard = enemyStartUmbral.board;
   735→   735→    triggerLogs = pushTriggerLogs(triggerLogs, freshState.turn, enemyStartUmbral);
   736→   736→
   737→   737→    // 2. Build the enemy hand: draw 3 random cards from a fresh shuffled pile.
   738→   738→    const resolvedDeck = resolveDeck(freshState);
   739→   739→    const enemyEnergy = freshState.maxEnergy;
   740→   740→    const enemyHand: HandCard[] = [];
   741→   741→    const deckPool = [...buildDrawPileFrom(resolvedDeck)];
   742→   742→    for (let i = 0; i < 3 && deckPool.length > 0; i++) {
   743→   743→      const idx = Math.floor(Math.random() * deckPool.length);
   744→   744→      const c = deckPool.splice(idx, 1)[0];
   745→   745→      enemyHand.push({ ...c, uid: uuidv4(), handBonus: 0 });
   746→   746→    }
   747→   747→
   748→   748→    // 3. Plan and execute the AI plays on the actual board, respecting lane
   749→   749→    //    bonuses AND lane traps. Use the post-trigger board as starting point.
   750→   750→    const plan = planEnemyTurn(
   751→   751→      triggerBoard,
   752→   752→      enemyHand,
   753→   753→      enemyEnergy,
   754→   754→      freshState.laneBonuses,
   755→   755→      freshState.laneTraps
   756→   756→    );
   757→   757→    let workBoard = triggerBoard;
   758→   758→    let workEnemyHand = [...enemyHand];
   759→   759→    let workEnergy = enemyEnergy;
   760→   760→    let workTraps: LaneTraps = { ...freshState.laneTraps };
   761→   761→    let log = triggerLogs;
   762→   762→    let gameStats: GameStats = { ...freshState.gameStats };
   763→   763→
   764→   764→    for (const play of plan) {
   765→   765→      try {
   766→   766→        const prePlayBoard = workBoard;
   767→   767→        const res = playCard(
   768→   768→          workBoard,
   769→   769→          workEnemyHand,
   770→   770→          workEnergy,
   771→   771→          "enemy",
   772→   772→          play.lane,
   773→   773→          play.card,
   774→   774→          freshState.laneBonuses[play.lane],
   775→   775→          workTraps[play.lane]
   776→   776→        );
   777→   777→        workBoard = res.board;
   778→   778→        workEnemyHand = res.hand;
   779→   779→        workEnergy = res.remainingEnergy;
   780→   780→
   781→   781→        // Update traps: clear if triggered, set if newTrap
   782→   782→        if (res.trapTriggered) {
   783→   783→          workTraps = { ...workTraps, [play.lane]: null };
   784→   784→        }
   785→   785→        if (res.newTrap) {
   786→   786→          workTraps = { ...workTraps, [play.lane]: res.newTrap };
   787→   787→        }
   788→   788→
   789→   789→        // Sound: enemy card play + buffs/debuffs/remove/trap
   790→   790→        if (useSettings.getState().soundOn) {
   791→   791→          sfx.cardPlay();
   792→   792→          if (res.mutations.some((m) => m.powerDelta > 0)) sfx.buff();
   793→   793→          if (res.mutations.some((m) => m.powerDelta < 0)) sfx.debuff();
   794→   794→          if (res.removedCards.length > 0) sfx.remove();
   795→   795→          if (res.trapTriggered) sfx.trapTrigger();
   796→   796→        }
   797→   797→
   798→   798→        // Enemy stats
   799→   799→        gameStats.cardsPlayedByEnemy++;
   800→   800→        gameStats.buffsApplied += res.mutations.filter(
   801→   801→          (m) => m.powerDelta > 0
   802→   802→        ).length;
   803→   803→        // debuffs to PLAYER cards count as "debuffs to enemy" from enemy's
   804→   804→        // perspective — we still keep a single "debuffsToEnemy" counter that
   805→   805→        // tracks debuffs applied TO the enemy side, so for enemy plays we
   806→   806→        // don't increment it (the enemy isn't debuffing itself).
   807→   807→        // (Player cards destroyed by enemy plays don't increment
   808→   808→        // `enemyCardsDestroyed` either — that stat is player-centric.)
   809→   809→
   810→   810→        log = pushLog(
   811→   811→          log,
   812→   812→          freshState.turn,
   813→   813→          "enemy",
   814→   814→          `Rival juega "${play.card.name}" en ${play.lane} (Poder ${res.playedCard.currentPower}).`,
   815→   815→          "play"
   816→   816→        );
   817→   817→        for (const m of res.mutations) {
   818→   818→          const sign = m.powerDelta > 0 ? "+" : "";
   819→   819→          log = pushLog(
   820→   820→            log,
   821→   821→            freshState.turn,
   822→   822→            m.side === "player" ? "player" : "enemy",
   823→   823→            `${sign}${m.powerDelta} en ${m.lane} (${m.side === "player" ? "tú" : "rival"}).`,
   824→   824→            m.powerDelta > 0 ? "buff" : "debuff"
   825→   825→          );
   826→   826→        }
   827→   827→        for (const rc of res.removedCards) {
   828→   828→          log = pushLog(
   829→   829→            log,
   830→   830→            freshState.turn,
   831→   831→            rc.side === "player" ? "player" : "enemy",
   832→   832→            `Carta destruida en ${rc.lane} (${rc.reason}).`,
   833→   833→            "debuff"
   834→   834→          );
   835→   835→        }
   836→   836→        if (res.trapTriggered) {
   837→   837→          log = pushLog(
   838→   838→            log,
   839→   839→            freshState.turn,
   840→   840→            "system",
   841→   841→            `¡Trampa enemiga activada en ${play.lane}!`,
   842→   842→            "debuff"
   843→   843→          );
   844→   844→        }
   845→   845→        if (res.newTrap) {
   846→   846→          log = pushLog(
   847→   847→            log,
   848→   848→            freshState.turn,
   849→   849→            "enemy",
   850→   850→            `Trampa enemiga colocada en ${play.lane}.`,
   851→   851→            "info"
   852→   852→          );
   853→   853→        }
   854→   854→
   855→   855→        // ====== TRIGGER DISPATCH (enemy play) ======
   856→   856→        // ON_DEATH: for each card destroyed by the enemy's play.
   857→   857→        // Look up the dying card in the PRE-play board (prePlayBoard) because
   858→   858→        // workBoard (= res.board) has already had removedCards filtered out.
   859→   859→        for (const rc of res.removedCards) {
   860→   860→          const deadCard = prePlayBoard[rc.lane][rc.side].find((c) => c.uid === rc.uid);
   861→   861→          if (!deadCard) continue;
   862→   862→          const deathRes = applyOnDeathEffects(workBoard, rc.side, rc.lane, deadCard);
   863→   863→          workBoard = deathRes.board;
   864→   864→          log = pushTriggerLogs(log, freshState.turn, deathRes);
   865→   865→
   866→   866→          // ====== ETAPA 4: emit card_destroyed event (enemy play) ======
   867→   867→          const destroyEvent: CardDestroyedEvent = {
   868→   868→            type: "card_destroyed",
   869→   869→            side: rc.side,
   870→   870→            lane: rc.lane,
   871→   871→            cardUid: rc.uid,
   872→   872→            cardName: deadCard.name,
   873→   873→            cardCategory: deadCard.category,
   874→   874→            reason: rc.reason,
   875→   875→          };
   876→   876→          const deathRxnRes = applyReactionEffects(workBoard, rc.side, destroyEvent);
   877→   877→          log = pushTriggerLogs(log, freshState.turn, deathRxnRes);
   878→   878→        }
   879→   879→        // UMBRAL: check after each enemy play.
   880→   880→        const enemyUmbralRes = checkUmbralTriggers(workBoard, freshState.hand);
   881→   881→        workBoard = enemyUmbralRes.board;
   882→   882→        log = pushTriggerLogs(log, freshState.turn, enemyUmbralRes);
   883→   883→
   884→   884→        // ====== ETAPA 4: emit card_played event (enemy play) ======
   885→   885→        // Reaction cards on the PLAYER side may respond to the enemy's card_played.
   886→   886→        // E.g. an ally Centinela could gain +X when the rival plays an Ataque card.
   887→   887→        // applyReactionEffects mutates workBoard in place.

   889→   889→          type: "card_played",
   890→   890→          side: "enemy",
   891→   891→          lane: play.lane,
   892→   892→          cardUid: res.playedCard.uid,
   893→   893→          cardName: play.card.name,
   894→   894→          cardCategory: play.card.category,
   895→   895→          cardSubtypes: play.card.subtypes,
   896→   896→        };
   897→   897→        const enemyRxnRes = applyReactionEffects(workBoard, "player", enemyPlayedEvent);
   898→   898→        log = pushTriggerLogs(log, freshState.turn, enemyRxnRes);
   899→   899→      } catch {
   900→   900→        // ignore failed plays
   901→   901→      }
   902→   902→    }
   903→   903→
   904→   904→    // ====== TRIGGER: END_TURN (enemy) ======
   905→   905→    // Enemy just finished playing — fire their end_turn effects before turn advances.
   906→   906→    const enemyEndRes = applyEndTurnEffects(workBoard, "enemy");
   907→   907→    workBoard = enemyEndRes.board;
   908→   908→    log = pushTriggerLogs(log, freshState.turn, enemyEndRes);
   909→   909→    const enemyEndUmbral = checkUmbralTriggers(workBoard, freshState.hand);
   910→   910→    workBoard = enemyEndUmbral.board;
   911→   911→    log = pushTriggerLogs(log, freshState.turn, enemyEndUmbral);
   912→   912→
   913→   913→    // 4. Advance turn, apply Sobrecarga energy penalty
   914→   914→    const newTurn = freshState.turn + 1;

  1104→  1104→  swapCardInDeck: (index, cardId) => {
  1105→  1105→    // Allow placing a card into ANY of the 12 deck slots, even if the slot is
  1106→  1106→    // beyond the current deckCardIds length (e.g. user emptied the deck then
  1107→  1107→    // clicks slot 5 — we pad with empty strings up to that index).
  1108→  1108→    if (index < 0 || index >= 12) return;
  1109→  1109→    const { deckCardIds } = get();
  1110→  1110→    if (deckCardIds[index] === cardId) return;
  1111→  1111→    // Prevent duplicates (deck is 12 unique cards) — if the card is already in
  1112→  1112→    // the deck at a DIFFERENT slot, move it from the old slot to the new one
  1113→  1113→    // (old slot becomes empty). This is the "drag-to-rearrange" UX.
  1114→  1114→    const next = [...deckCardIds];
  1115→  1115→    // Pad with empty strings up to the target index so we can place at any slot
  1116→  1116→    while (next.length <= index) next.push("");
  1117→  1117→    if (deckCardIds.includes(cardId)) {
  1118→  1118→      const existingIdx = deckCardIds.indexOf(cardId);
  1119→  1119→      next[existingIdx] = "";  // clear old slot
  1120→  1120→    }
  1121→  1121→    next[index] = cardId;
  1122→  1122→    // Always store exactly 12 entries (pad/truncate to 12)
  1123→  1123→    while (next.length < 12) next.push("");
  1124→  1124→    set({ deckCardIds: next.slice(0, 12), deckDirty: true, deckSaveStatus: "idle" });
  1125→  1125→  },
